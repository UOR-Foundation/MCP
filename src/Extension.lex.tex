definition{
  name: callAnswer,
  parameters: [answered : (Result CallOutcome Refusal::ProviderRefusal)],
  result: (Result Methods::ResultBody Refusal::ProviderRefusal),
  body: (match answered
    | Result.ok outcome => (match outcome
      | CallOutcome.complete found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Methods::ResultBody.callTool found))
      | CallOutcome.pending needed => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Methods::ResultBody.inputRequired needed))
      | CallOutcome.task created => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Methods::ResultBody.taskCreated created))
      | CallOutcome.perform request => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Refusal::ProviderRefusal.effectRequested request)))
    | Result.error pr => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] pr)),
  executable: true
}

definition{
  name: answerOf,
  parameters: [provider : Provider, request : Methods::ClientRequest],
  result: (Result Methods::ResultBody Refusal::ProviderRefusal),
  body: (match request
    | Methods::ClientRequest.serverDiscover params => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Refusal::ProviderRefusal.internal "server/discover is answered by the core"))
    | Methods::ClientRequest.toolsList params => (match (apply (project provider listTools) params)
      | Result.ok found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Methods::ResultBody.listTools found))
      | Result.error pr => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] pr))
    | Methods::ClientRequest.toolsCall params => (callAnswer (apply (project provider callTool) params))
    | Methods::ClientRequest.resourcesList params => (match (apply (project provider listResources) params)
      | Result.ok found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Methods::ResultBody.listResources found))
      | Result.error pr => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] pr))
    | Methods::ClientRequest.resourcesRead params => (readAnswer (apply (project provider readResource) params))
    | Methods::ClientRequest.resourcesTemplatesList params => (match (apply (project provider listResourceTemplates) params)
      | Result.ok found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Methods::ResultBody.listResourceTemplates found))
      | Result.error pr => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] pr))
    | Methods::ClientRequest.promptsList params => (match (apply (project provider listPrompts) params)
      | Result.ok found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Methods::ResultBody.listPrompts found))
      | Result.error pr => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] pr))
    | Methods::ClientRequest.promptsGet params => (promptAnswer (apply (project provider getPrompt) params))
    | Methods::ClientRequest.completionComplete params => (match (apply (project provider complete) params)
      | Result.ok found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Methods::ResultBody.complete found))
      | Result.error pr => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] pr))
    | Methods::ClientRequest.tasksGet params => (match (apply (project provider getTask) params)
      | Result.ok found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Methods::ResultBody.getTask found))
      | Result.error pr => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] pr))
    | Methods::ClientRequest.tasksUpdate params => (match (apply (project provider updateTask) params)
      | Option.none => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Methods::ResultBody.acknowledged))
      | Option.some pr => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] pr))
    | Methods::ClientRequest.tasksCancel params => (match (apply (project provider cancelTask) params)
      | Option.none => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Methods::ResultBody.acknowledged))
      | Option.some pr => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] pr)))
}
