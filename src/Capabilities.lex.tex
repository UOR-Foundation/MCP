definition{
  name: isDeclared,
  parameters: [capability : (Option Json::JsonObject)],
  result: Bool,
  body: (match capability | Option.none => false | Option.some declared => true),
  executable: true
}

definition{
  name: declaresExtension,
  parameters: [extensions : (Option Json::JsonObject), identifier : String],
  result: Bool,
  body: (match extensions
    | Option.none => false
    | Option.some declared => (Json::hasKey identifier declared)),
  executable: true
}

definition{
  name: clientDeclaresTasks,
  parameters: [capabilities : ClientCapabilities],
  result: Bool,

  executable: true
}
