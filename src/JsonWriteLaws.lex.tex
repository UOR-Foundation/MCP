definition{
  name: jsonsOf,
  parameters: [trees : Forest],
  result: (List Json::Json),
  body: (match trees
    | Forest.stop => (nil Json::Json)
    | Forest.null label next => (cons (ctor Json::Json.null) (jsonsOf next))
    | Forest.bool label flag next => (cons (ctor Json::Json.bool flag) (jsonsOf next))
    | Forest.number label amount next => (cons (ctor Json::Json.number amount) (jsonsOf next))
    | Forest.text label content next => (cons (ctor Json::Json.text content) (jsonsOf next))
    | Forest.array label kids next => (cons (ctor Json::Json.array (jsonsOf kids)) (jsonsOf next))
    | Forest.object label kids next => (cons (ctor Json::Json.object (membersOf kids)) (jsonsOf next))),
  mutual: Spread,
  recursive_argument: trees
}

definition{
  name: membersOf,
  parameters: [trees : Forest],
  result: (List (String * Json::Json)),
  body: (match trees
    | Forest.stop => (nil (String * Json::Json))
    | Forest.null label next => (cons ((keyOf label), (ctor Json::Json.null)) (membersOf next))
    | Forest.bool label flag next => (cons ((keyOf label), (ctor Json::Json.bool flag)) (membersOf next))
    | Forest.number label amount next => (cons ((keyOf label), (ctor Json::Json.number amount)) (membersOf next))
    | Forest.text label content next => (cons ((keyOf label), (ctor Json::Json.text content)) (membersOf next))

    | Forest.object label kids next => (cons ((keyOf label), (ctor Json::Json.object (membersOf kids))) (membersOf next))),
  mutual: Spread,
  recursive_argument: trees

    | Phase.afterItem => (record Outline {bytes := (cons 44 (cat (project value bytes) (project rest bytes))), cost := ((((plus (project rest cost) (project value cost)) + 1) + 1) + 1), failure := (orElse (project value failure) (project rest failure))})
    | Phase.entries => (record Outline {bytes := (cat (prim append (JsonWrite::quoted key) (cons 58 (nil Nat)) : (List Nat)) (cat (project value bytes) (project rest bytes))), cost := ((plus (project rest cost) ((project value cost) + 1)) + 1), failure := (orElse (project value failure) (project rest failure))})
    | Phase.afterEntry => (record Outline {bytes := (cons 44 (cat (prim append (JsonWrite::quoted key) (cons 58 (nil Nat)) : (List Nat)) (cat (project value bytes) (project rest bytes)))), cost := ((((plus (project rest cost) ((project value cost) + 1)) + 1) + 1) + 1), failure := (orElse (project value failure) (project rest failure))})),

definition{
  name: appendForest,
  parameters: [lead : Forest, trail : Forest],
  result: Forest,
  body: (match lead
    | Forest.stop => trail
    | Forest.null label next => (ctor Forest.null label (appendForest next trail))
    | Forest.bool label flag next => (ctor Forest.bool label flag (appendForest next trail))
    | Forest.number label amount next => (ctor Forest.number label amount (appendForest next trail))
    | Forest.text label content next => (ctor Forest.text label content (appendForest next trail))
    | Forest.array label kids next => (ctor Forest.array label kids (appendForest next trail))
    | Forest.object label kids next => (ctor Forest.object label kids (appendForest next trail))),
  recursive_argument: lead
}

definition{
  name: relabel,
  parameters: [key : String, trees : Forest],
  result: Forest,
  body: (match trees
    | Forest.stop => (ctor Forest.stop)
    | Forest.null label next => (ctor Forest.null (ctor Option.some [String] key) next)
    | Forest.bool label flag next => (ctor Forest.bool (ctor Option.some [String] key) flag next)
    | Forest.number label amount next => (ctor Forest.number (ctor Option.some [String] key) amount next)
    | Forest.text label content next => (ctor Forest.text (ctor Option.some [String] key) content next)
    | Forest.array label kids next => (ctor Forest.array (ctor Option.some [String] key) kids next)
    | Forest.object label kids next => (ctor Forest.object (ctor Option.some [String] key) kids next))
}

definition{
  name: appendJsons,
  parameters: [left : (List Json::Json), right : (List Json::Json)],
  result: (List Json::Json),
  body: (match left
    | List.nil => right
    | List.cons head tail => (cons head (appendJsons tail right))),
  recursive_argument: left
}

definition{
  name: appendMembers,
  parameters: [left : (List (String * Json::Json)), right : (List (String * Json::Json))],
  result: (List (String * Json::Json)),
  body: (match left
    | List.nil => right
    | List.cons head tail => (cons head (appendMembers tail right))),
  recursive_argument: left
}

definition{
  name: relabelHead,
  parameters: [key : String, entries : (List (String * Json::Json))],
  result: (List (String * Json::Json)),
  body: (match entries
    | List.nil => (nil (String * Json::Json))
    | List.cons head tail => (match head | Prod.mk old value => (cons (key, value) tail)))
}

definition{
  name: toForest,
  parameters: [value : Json::Json],
  result: Forest,
  body: (match value
    | Json::Json.null => (ctor Forest.null (ctor Option.none [String]) (ctor Forest.stop))
    | Json::Json.bool flag => (ctor Forest.bool (ctor Option.none [String]) flag (ctor Forest.stop))
    | Json::Json.number amount => (ctor Forest.number (ctor Option.none [String]) amount (ctor Forest.stop))
    | Json::Json.text content => (ctor Forest.text (ctor Option.none [String]) content (ctor Forest.stop))
    | Json::Json.array items => (ctor Forest.array (ctor Option.none [String]) (toForestList items) (ctor Forest.stop))
    | Json::Json.object entries => (ctor Forest.object (ctor Option.none [String]) (toForestMembers entries) (ctor Forest.stop))),
  mutual: Encode,
  recursive_argument: value
}

definition{
  name: toForestList,
  parameters: [items : (List Json::Json)],
  result: Forest,
  body: (match items
    | List.nil => (ctor Forest.stop)
    | List.cons head tail => (appendForest (toForest head) (toForestList tail))),
  mutual: Encode,
  recursive_argument: items
}

definition{
  name: toForestMembers,
  parameters: [entries : (List (String * Json::Json))],
  result: Forest,
  body: (match entries
    | List.nil => (ctor Forest.stop)
    | List.cons entry tail => (appendForest (toForestPair entry) (toForestMembers tail))),
  mutual: Encode,
  recursive_argument: entries
}

definition{
  name: toForestPair,
  parameters: [entry : (String * Json::Json)],
  result: Forest,
  body: (match entry | Prod.mk key value => (relabel key (toForest value))),
  mutual: Encode,
  recursive_argument: entry
}
