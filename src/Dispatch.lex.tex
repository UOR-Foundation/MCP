definition{
  name: isOffered,
  type_parameters: [T],
  parameters: [capability : (Option 'T)],
  result: Bool,
  body: (match capability | Option.none => false | Option.some declared => true)
}

definition{
  name: offers,
  parameters: [capabilities : Capabilities::ServerCapabilities, method : Methods::Method],
  result: Bool,
  body: (match method
    | Methods::Method.serverDiscover => true
    | Methods::Method.toolsList => (isOffered [Capabilities::ListChangedCapability] (project capabilities tools))
    | Methods::Method.toolsCall => (isOffered [Capabilities::ListChangedCapability] (project capabilities tools))
    | Methods::Method.resourcesList => (isOffered [Capabilities::ResourcesCapability] (project capabilities resources))
    | Methods::Method.resourcesRead => (isOffered [Capabilities::ResourcesCapability] (project capabilities resources))
    | Methods::Method.resourcesTemplatesList => (isOffered [Capabilities::ResourcesCapability] (project capabilities resources))
    | Methods::Method.promptsList => (isOffered [Capabilities::ListChangedCapability] (project capabilities prompts))
    | Methods::Method.promptsGet => (isOffered [Capabilities::ListChangedCapability] (project capabilities prompts))
    | Methods::Method.completionComplete => (Capabilities::isDeclared (project capabilities completions))
    | Methods::Method.tasksGet => (Capabilities::declaresExtension (project capabilities extensions) (Capabilities::tasksExtensionId))
    | Methods::Method.tasksUpdate => (Capabilities::declaresExtension (project capabilities extensions) (Capabilities::tasksExtensionId))
    | Methods::Method.tasksCancel => (Capabilities::declaresExtension (project capabilities extensions) (Capabilities::tasksExtensionId)))
}

definition{
  name: replyOf,
  parameters: [provider : Extension::Provider, body : Methods::ResultBody],
  result: Reply,
  body: (record Reply {body := body, meta := (record Meta::ResultMeta {serverInfo := (ctor Option.some [Common::Implementation] (project provider serverInfo)), others := (ctor Option.none [Json::JsonObject])})})
}

definition{
  name: pendingGate,
  parameters: [provider : Extension::Provider, needed : Input::InputRequiredResult, covered : Bool],
  result: (Result Reply Refusal::Refusal),
  body: (if covered then (ctor Result.ok [Reply, Refusal::Refusal] (replyOf provider (ctor Methods::ResultBody.inputRequired needed))) else (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.provider (ctor Refusal::ProviderRefusal.missingCapability (Input::requiredCapabilities (Input::requestsOf needed))))))
}

definition{
  name: pendingReply,
  parameters: [
    provider : Extension::Provider,
    meta : Meta::RequestMeta,
    needed : Input::InputRequiredResult
  ],
  result: (Result Reply Refusal::Refusal),
  body: (pendingGate provider needed (Input::covers (project meta clientCapabilities) (Input::requestsOf needed)))
}

definition{
  name: taskGate,
  parameters: [provider : Extension::Provider, created : Tasks::CreateTaskResult, declared : Bool],
  result: (Result Reply Refusal::Refusal),
  body: (if declared then (ctor Result.ok [Reply, Refusal::Refusal] (replyOf provider (ctor Methods::ResultBody.taskCreated created))) else (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.provider (ctor Refusal::ProviderRefusal.missingCapability (Capabilities::tasksRequirement)))))
}

definition{
  name: taskReply,
  parameters: [
    provider : Extension::Provider,
    meta : Meta::RequestMeta,
    created : Tasks::CreateTaskResult
  ],
  result: (Result Reply Refusal::Refusal),
  body: (taskGate provider created (Capabilities::clientDeclaresTasks (project meta clientCapabilities)))
}

definition{
  name: run,
  parameters: [provider : Extension::Provider, request : Methods::ClientRequest],
  result: (Result Reply Refusal::Refusal),
  body: (match request
    | Methods::ClientRequest.serverDiscover params => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf provider (ctor Methods::ResultBody.discover (record Methods::DiscoverResult {cache := (project provider discoverCache), supportedVersions := (Validation::supportedVersions), capabilities := (project provider capabilities), instructions := (project provider instructions)}))))
    | Methods::ClientRequest.toolsList params => (match (apply (project provider listTools) params)
      | Result.ok found => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf provider (ctor Methods::ResultBody.listTools found)))
      | Result.error pr => (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.provider pr)))
    | Methods::ClientRequest.resourcesList params => (match (apply (project provider listResources) params)
      | Result.ok found => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf provider (ctor Methods::ResultBody.listResources found)))
      | Result.error pr => (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.provider pr)))
    | Methods::ClientRequest.resourcesTemplatesList params => (match (apply (project provider listResourceTemplates) params)
      | Result.ok found => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf provider (ctor Methods::ResultBody.listResourceTemplates found)))
      | Result.error pr => (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.provider pr)))
    | Methods::ClientRequest.promptsList params => (match (apply (project provider listPrompts) params)
      | Result.ok found => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf provider (ctor Methods::ResultBody.listPrompts found)))
      | Result.error pr => (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.provider pr)))
    | Methods::ClientRequest.completionComplete params => (match (apply (project provider complete) params)
      | Result.ok found => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf provider (ctor Methods::ResultBody.complete found)))
      | Result.error pr => (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.provider pr)))
    | Methods::ClientRequest.tasksGet params => (match (apply (project provider getTask) params)
      | Result.ok found => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf provider (ctor Methods::ResultBody.getTask found)))
      | Result.error pr => (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.provider pr)))
    | Methods::ClientRequest.toolsCall params => (match (apply (project provider callTool) params)
      | Result.ok outcome => (match outcome
        | Extension::CallOutcome.complete found => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf provider (ctor Methods::ResultBody.callTool found)))
        | Extension::CallOutcome.pending needed => (pendingReply provider (project params meta) needed)
        | Extension::CallOutcome.task created => (taskReply provider (project params meta) created))
      | Result.error pr => (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.provider pr)))
    | Methods::ClientRequest.resourcesRead params => (match (apply (project provider readResource) params)
      | Result.ok outcome => (match outcome
        | Extension::ReadOutcome.complete found => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf provider (ctor Methods::ResultBody.readResource found)))
        | Extension::ReadOutcome.pending needed => (pendingReply provider (project params meta) needed))
      | Result.error pr => (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.provider pr)))
    | Methods::ClientRequest.promptsGet params => (match (apply (project provider getPrompt) params)
      | Result.ok outcome => (match outcome
        | Extension::PromptOutcome.complete found => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf provider (ctor Methods::ResultBody.getPrompt found)))
        | Extension::PromptOutcome.pending needed => (pendingReply provider (project params meta) needed))
      | Result.error pr => (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.provider pr)))
    | Methods::ClientRequest.tasksUpdate params => (match (apply (project provider updateTask) params)
      | Option.none => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf provider (ctor Methods::ResultBody.acknowledged)))
      | Option.some pr => (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.provider pr)))
    | Methods::ClientRequest.tasksCancel params => (match (apply (project provider cancelTask) params)
      | Option.none => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf provider (ctor Methods::ResultBody.acknowledged)))
      | Option.some pr => (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.provider pr))))
}

definition{
  name: afterDecode,
  parameters: [
    provider : Extension::Provider,
    decoded : (Result Methods::ClientRequest Refusal::ParamsProblem)
  ],
  result: (Result Reply Refusal::Refusal),
  body: (match decoded
    | Result.error problem => (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.invalidParams problem))
    | Result.ok typed => (run provider typed))
}

definition{
  name: gateOffer,
  parameters: [provider : Extension::Provider, routed : Validation::Routed, offered : Bool],
  result: (Result Reply Refusal::Refusal),
  body: (if offered then (afterDecode provider (Decode::decodeParams (project routed method) (project routed meta) (project routed params))) else (ctor Result.error [Reply, Refusal::Refusal] (ctor Refusal::Refusal.methodNotFound (Methods::methodWire (project routed method)))))
}

definition{
  name: dispatch,
  parameters: [provider : Extension::Provider, routed : Validation::Routed],
  result: (Result Reply Refusal::Refusal),
  body: (gateOffer provider routed (offers (project provider capabilities) (project routed method)))
}

definition{
  name: afterValidation,
  parameters: [
    provider : Extension::Provider,
    validated : (Result Validation::Routed Refusal::Refusal)
  ],
  result: (Result Reply Refusal::Refusal),
  body: (match validated
    | Result.error refusal => (ctor Result.error [Reply, Refusal::Refusal] refusal)
    | Result.ok routed => (dispatch provider routed))
}

definition{
  name: isMethodNotFoundRefusal,
  parameters: [refusal : Refusal::Refusal],
  result: Bool,
  body: (match refusal
    | Refusal::Refusal.invalidParams problem => false
    | Refusal::Refusal.unsupportedProtocolVersion requested supported => false
    | Refusal::Refusal.headerMismatch problem => false
    | Refusal::Refusal.methodNotFound method => true
    | Refusal::Refusal.provider inner => false)
}

definition{
  name: refusesUnknown,
  parameters: [route : Methods::MethodRoute, outcome : (Result Reply Refusal::Refusal)],
  result: Bool,
  body: (match route
    | Methods::MethodRoute.known method => true
    | Methods::MethodRoute.unknown => (match outcome
      | Result.ok reply => false
      | Result.error refusal => true))
}

definition{
  name: isDiscoverMethod,
  parameters: [method : Methods::Method],
  result: Bool,
  body: (match method
    | Methods::Method.serverDiscover => true
    | Methods::Method.toolsList => false
    | Methods::Method.toolsCall => false
    | Methods::Method.resourcesList => false
    | Methods::Method.resourcesRead => false
    | Methods::Method.resourcesTemplatesList => false
    | Methods::Method.promptsList => false
    | Methods::Method.promptsGet => false
    | Methods::Method.completionComplete => false
    | Methods::Method.tasksGet => false
    | Methods::Method.tasksUpdate => false
    | Methods::Method.tasksCancel => false)
}

definition{
  name: discoverOrRefused,
  parameters: [method : Methods::Method, outcome : (Result Reply Refusal::Refusal)],
  result: Bool,
  body: (if (isDiscoverMethod method) then true else (match outcome
    | Result.ok reply => false
    | Result.error refusal => (isMethodNotFoundRefusal refusal)))
}

definition{
  name: codeOfHandled,
  parameters: [handled : Handled],
  result: (Option Int),
  body: (match (project handled outcome)
    | Result.ok reply => (ctor Option.none [Int])
    | Result.error refusal => (ctor Option.some [Int] (Refusal::codeOf refusal)))
}

definition{
  name: typeOfHandled,
  parameters: [handled : Handled],
  result: (Option Meta::ResultType),
  body: (match (project handled outcome)
    | Result.ok reply => (ctor Option.some [Meta::ResultType] (Methods::resultTypeOf (project reply body)))
    | Result.error refusal => (ctor Option.none [Meta::ResultType]))
}

definition{
  name: serverNameOfHandled,
  parameters: [handled : Handled],

  body: (match (project handled outcome)
    | Result.ok reply => (match (project (project reply meta) serverInfo)
      | Option.none => (ctor Option.none [String])
      | Option.some info => (ctor Option.some [String] (project info name)))
    | Result.error refusal => (ctor Option.none [String]))
