definition{
  name: offers,
  parameters: [capabilities : Capabilities::ServerCapabilities, method : Methods::Method],
  result: Bool,
  body: (match method
    | Methods::Method.serverDiscover => true
    | Methods::Method.toolsList => (isOffered [Capabilities::ListChangedCapability] (project capabilities tools))
    | Methods::Method.toolsCall => (isOffered [Capabilities::ListChangedCapability] (project capabilities tools))
    | Methods::Method.resourcesList => (isOffered [Capabilities::ResourcesCapability] (project capabilities resources))
    | Methods::Method.resourcesRead => (isOffered [Capabilities::ResourcesCapability] (project capabilities resources))
    | Methods::Method.resourcesTemplatesList => (isOffered [Capabilities::ResourcesCapability] (project capabilities resources))
    | Methods::Method.promptsList => (isOffered [Capabilities::ListChangedCapability] (project capabilities prompts))
    | Methods::Method.promptsGet => (isOffered [Capabilities::ListChangedCapability] (project capabilities prompts))
    | Methods::Method.completionComplete => (Capabilities::isDeclared (project capabilities completions))
    | Methods::Method.tasksGet => (Capabilities::declaresExtension (project capabilities extensions) (Capabilities::tasksExtensionId))
    | Methods::Method.tasksUpdate => (Capabilities::declaresExtension (project capabilities extensions) (Capabilities::tasksExtensionId))
    | Methods::Method.tasksCancel => (Capabilities::declaresExtension (project capabilities extensions) (Capabilities::tasksExtensionId))),
  executable: true

  parameters: [method : Methods::Method, body : Methods::ResultBody],
  result: Bool,
  body: (match body
    | Methods::ResultBody.discover found => (sameMethod method (ctor Methods::Method.serverDiscover))
    | Methods::ResultBody.listTools found => (sameMethod method (ctor Methods::Method.toolsList))
    | Methods::ResultBody.callTool found => (sameMethod method (ctor Methods::Method.toolsCall))
    | Methods::ResultBody.listResources found => (sameMethod method (ctor Methods::Method.resourcesList))
    | Methods::ResultBody.readResource found => (sameMethod method (ctor Methods::Method.resourcesRead))
    | Methods::ResultBody.listResourceTemplates found => (sameMethod method (ctor Methods::Method.resourcesTemplatesList))
    | Methods::ResultBody.listPrompts found => (sameMethod method (ctor Methods::Method.promptsList))

    | Methods::ResultBody.getTask found => (sameMethod method (ctor Methods::Method.tasksGet))
    | Methods::ResultBody.acknowledged => ((sameMethod method (ctor Methods::Method.tasksUpdate)) || (sameMethod method (ctor Methods::Method.tasksCancel)))

    | Methods::ResultBody.taskCreated created => (Methods::supportsTaskResult method)),

definition{

  parameters: [
    info : Extension::ProviderInfo,
    request : Methods::ClientRequest,
    body : Methods::ResultBody

    | Methods::ResultBody.inputRequired needed => (pendingReply info (metaOfRequest request) needed)

    | Methods::ResultBody.callTool found => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf info body))
    | Methods::ResultBody.listResources found => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf info body))

    | Methods::ResultBody.complete found => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf info body))
    | Methods::ResultBody.getTask found => (ctor Result.ok [Reply, Refusal::Refusal] (replyOf info body))

definition{
  name: isMethodNotFoundRefusal,
  parameters: [refusal : Refusal::Refusal],
  result: Bool,
  body: (match refusal
    | Refusal::Refusal.invalidParams problem => false
    | Refusal::Refusal.unsupportedProtocolVersion requested supported => false
    | Refusal::Refusal.headerMismatch problem => false
    | Refusal::Refusal.methodNotFound method => true
    | Refusal::Refusal.provider inner => false)
}

definition{
  name: refusesUnknown,
  parameters: [route : Methods::MethodRoute, outcome : (Result Reply Refusal::Refusal)],
  result: Bool,
  body: (match route
    | Methods::MethodRoute.known method => true
    | Methods::MethodRoute.unknown => (match outcome
      | Result.ok reply => false
      | Result.error refusal => true))
}

definition{
  name: discoverOrRefused,
  parameters: [method : Methods::Method, outcome : (Result Reply Refusal::Refusal)],
  result: Bool,
  body: (if (isDiscoverMethod method) then true else (match outcome
    | Result.ok reply => false
    | Result.error refusal => (isMethodNotFoundRefusal refusal)))
}
