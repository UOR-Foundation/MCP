definition{
  name: isSupported,
  parameters: [version : String, versions : (List String)],
  result: Bool,
  body: (match versions
    | List.nil => false
    | List.cons head tail => (if (prim equal head version : Bool) then true else (isSupported version tail))),
  recursive_argument: versions
}

definition{
  name: optionalCapability,
  parameters: [key : String, object : Json::JsonObject],
  result: (Option (Option Json::JsonObject)),
  body: (match (Json::lookup key object)
    | Json::Lookup.absent => (ctor Option.some [(Option Json::JsonObject)] (ctor Option.none [Json::JsonObject]))
    | Json::Lookup.present value => (match (Json::asObject value)
      | Option.none => (ctor Option.none [(Option Json::JsonObject)])
      | Option.some found => (ctor Option.some [(Option Json::JsonObject)] (ctor Option.some [Json::JsonObject] found))))
}

definition{
  name: decodeCapabilities,
  parameters: [value : Json::Json],
  result: (Option Capabilities::ClientCapabilities),
  body: (match (Json::asObject value)
    | Option.none => (ctor Option.none [Capabilities::ClientCapabilities])
    | Option.some object => (match (optionalCapability "experimental" object)
      | Option.none => (ctor Option.none [Capabilities::ClientCapabilities])
      | Option.some experimental => (match (optionalCapability "roots" object)
        | Option.none => (ctor Option.none [Capabilities::ClientCapabilities])
        | Option.some roots => (match (optionalCapability "sampling" object)
          | Option.none => (ctor Option.none [Capabilities::ClientCapabilities])
          | Option.some sampling => (match (optionalCapability "elicitation" object)
            | Option.none => (ctor Option.none [Capabilities::ClientCapabilities])
            | Option.some elicitation => (match (optionalCapability "extensions" object)
              | Option.none => (ctor Option.none [Capabilities::ClientCapabilities])
              | Option.some extensions => (ctor Option.some [Capabilities::ClientCapabilities] (record Capabilities::ClientCapabilities {experimental := experimental, roots := roots, sampling := sampling, elicitation := elicitation, extensions := extensions}))))))))
}

definition{
  name: readLogLevel,
  parameters: [object : Json::JsonObject],
  result: (Result (Option Common::LoggingLevel) Refusal::ParamsProblem),
  body: (match (Decode::optText (Meta::logLevelKey) object)
    | Result.error problem => (ctor Result.error [(Option Common::LoggingLevel), Refusal::ParamsProblem] problem)
    | Result.ok wire => (match wire
      | Option.none => (ctor Result.ok [(Option Common::LoggingLevel), Refusal::ParamsProblem] (ctor Option.none [Common::LoggingLevel]))
      | Option.some text => (match (Common::loggingLevelOfWire text)
        | Common::LoggingLevelRead.recognized level => (ctor Result.ok [(Option Common::LoggingLevel), Refusal::ParamsProblem] (ctor Option.some [Common::LoggingLevel] level))
        | Common::LoggingLevelRead.unrecognized => (ctor Result.error [(Option Common::LoggingLevel), Refusal::ParamsProblem] (ctor Refusal::ParamsProblem.malformedField (Meta::logLevelKey))))))
}

definition{
  name: readProgressToken,
  parameters: [object : Json::JsonObject],
  result: (Result (Option Common::ProgressToken) Refusal::ParamsProblem),
  body: (match (Json::lookup (Meta::progressTokenKey) object)
    | Json::Lookup.absent => (ctor Result.ok [(Option Common::ProgressToken), Refusal::ParamsProblem] (ctor Option.none [Common::ProgressToken]))
    | Json::Lookup.present value => (match (Json::asText value)
      | Option.some content => (ctor Result.ok [(Option Common::ProgressToken), Refusal::ParamsProblem] (ctor Option.some [Common::ProgressToken] (ctor Common::ProgressToken.text content)))
      | Option.none => (match (Json::asInteger value)
        | Option.some whole => (ctor Result.ok [(Option Common::ProgressToken), Refusal::ParamsProblem] (ctor Option.some [Common::ProgressToken] (ctor Common::ProgressToken.number whole)))
        | Option.none => (ctor Result.error [(Option Common::ProgressToken), Refusal::ParamsProblem] (ctor Refusal::ParamsProblem.malformedField (Meta::progressTokenKey)))))),
  axioms: [propext]
}

definition{
  name: buildMeta,
  parameters: [meta : Json::JsonObject, version : Json::Json, capabilities : Json::Json],
  result: (Result Meta::RequestMeta Refusal::ParamsProblem),
  body: (match (Json::asText version)
    | Option.none => (ctor Result.error [Meta::RequestMeta, Refusal::ParamsProblem] (ctor Refusal::ParamsProblem.malformedProtocolVersion))
    | Option.some protocolVersion => (match (decodeCapabilities capabilities)
      | Option.none => (ctor Result.error [Meta::RequestMeta, Refusal::ParamsProblem] (ctor Refusal::ParamsProblem.malformedClientCapabilities))
      | Option.some clientCapabilities => (match (Decode::optObject (Meta::clientInfoKey) meta)
        | Result.error problem => (ctor Result.error [Meta::RequestMeta, Refusal::ParamsProblem] problem)
        | Result.ok clientInfo => (match (readLogLevel meta)
          | Result.error problem => (ctor Result.error [Meta::RequestMeta, Refusal::ParamsProblem] problem)
          | Result.ok logLevel => (match (readProgressToken meta)
            | Result.error problem => (ctor Result.error [Meta::RequestMeta, Refusal::ParamsProblem] problem)
            | Result.ok progressToken => (ctor Result.ok [Meta::RequestMeta, Refusal::ParamsProblem] (record Meta::RequestMeta {protocolVersion := protocolVersion, clientCapabilities := clientCapabilities, clientInfo := clientInfo, logLevel := logLevel, progressToken := progressToken, raw := meta}))))))),
  axioms: [propext]
}

definition{
  name: metaFromLookups,
  parameters: [meta : Json::JsonObject, version : Json::Lookup, capabilities : Json::Lookup],
  result: (Result Meta::RequestMeta Refusal::ParamsProblem),
  body: (match version
    | Json::Lookup.absent => (ctor Result.error [Meta::RequestMeta, Refusal::ParamsProblem] (ctor Refusal::ParamsProblem.missingProtocolVersion))
    | Json::Lookup.present versionValue => (match capabilities
      | Json::Lookup.absent => (ctor Result.error [Meta::RequestMeta, Refusal::ParamsProblem] (ctor Refusal::ParamsProblem.missingClientCapabilities))
      | Json::Lookup.present capabilitiesValue => (buildMeta meta versionValue capabilitiesValue))),
  axioms: [propext]
}

    | KeyPresence.bothPresent => (metaFromLookups meta (Json::lookup (Meta::protocolVersionKey) meta) (Json::lookup (Meta::clientCapabilitiesKey) meta))),

definition{
  name: metaOf,
  parameters: [params : (Option Json::JsonObject)],
  result: (Result Meta::RequestMeta Refusal::ParamsProblem),
  body: (match params
    | Option.none => (ctor Result.error [Meta::RequestMeta, Refusal::ParamsProblem] (ctor Refusal::ParamsProblem.missingParams))
    | Option.some object => (match (Json::lookup "_meta" object)
      | Json::Lookup.absent => (ctor Result.error [Meta::RequestMeta, Refusal::ParamsProblem] (ctor Refusal::ParamsProblem.missingMeta))
      | Json::Lookup.present value => (match (Json::asObject value)
        | Option.none => (ctor Result.error [Meta::RequestMeta, Refusal::ParamsProblem] (ctor Refusal::ParamsProblem.malformedMeta))
        | Option.some meta => (metaFromObject meta)))),
  axioms: [propext]
}

definition{

    | Methods::NameField.noName => (ctor Option.none [String])
    | Methods::NameField.name => (match (Json::lookup "name" params)
      | Json::Lookup.absent => (ctor Option.none [String])
      | Json::Lookup.present value => (Json::asText value))
    | Methods::NameField.uri => (match (Json::lookup "uri" params)
      | Json::Lookup.absent => (ctor Option.none [String])
      | Json::Lookup.present value => (Json::asText value))
    | Methods::NameField.taskId => (match (Json::lookup "taskId" params)
      | Json::Lookup.absent => (ctor Option.none [String])
      | Json::Lookup.present value => (Json::asText value)))

definition{
  name: nameHeaderAgainst,
  parameters: [header : (Option String), body : (Option String)],
  result: (Option Refusal::HeaderProblem),
  body: (match header
    | Option.none => (ctor Option.some [Refusal::HeaderProblem] (ctor Refusal::HeaderProblem.missingNameHeader))
    | Option.some value => (match body
      | Option.none => (ctor Option.none [Refusal::HeaderProblem])
      | Option.some expected => (if (prim equal value expected : Bool) then (ctor Option.none [Refusal::HeaderProblem]) else (ctor Option.some [Refusal::HeaderProblem] (ctor Refusal::HeaderProblem.nameDiffers)))))
}

definition{
  name: nameHeaderCheck,
  parameters: [header : (Option String), field : Methods::NameField, params : Json::JsonObject],
  result: (Option Refusal::HeaderProblem),
  body: (match field
    | Methods::NameField.noName => (ctor Option.none [Refusal::HeaderProblem])
    | Methods::NameField.name => (nameHeaderAgainst header (nameInBody field params))
    | Methods::NameField.uri => (nameHeaderAgainst header (nameInBody field params))
    | Methods::NameField.taskId => (nameHeaderAgainst header (nameInBody field params)))
}

definition{
  name: headerCheck,
  parameters: [
    headers : (Option Headers),
    method : Methods::Method,
    wireMethod : String,
    version : String,
    params : Json::JsonObject
  ],
  result: (Option Refusal::HeaderProblem),
  body: (match headers
    | Option.none => (ctor Option.none [Refusal::HeaderProblem])
    | Option.some present => (match (project present protocolVersion)
      | Option.none => (ctor Option.some [Refusal::HeaderProblem] (ctor Refusal::HeaderProblem.missingProtocolVersionHeader))
      | Option.some headerVersion => (if (prim equal headerVersion version : Bool) then (match (project present method)
        | Option.none => (ctor Option.some [Refusal::HeaderProblem] (ctor Refusal::HeaderProblem.missingMethodHeader))
        | Option.some headerMethod => (if (prim equal headerMethod wireMethod : Bool) then (nameHeaderCheck (project present name) (Methods::nameFieldOf method) params) else (ctor Option.some [Refusal::HeaderProblem] (ctor Refusal::HeaderProblem.methodDiffers)))) else (ctor Option.some [Refusal::HeaderProblem] (ctor Refusal::HeaderProblem.protocolVersionDiffers)))))
}

definition{
  name: gate,
  parameters: [supported : Bool, version : String, versions : (List String)],
  result: (Option Refusal::Refusal),
  body: (if supported then (ctor Option.none [Refusal::Refusal]) else (ctor Option.some [Refusal::Refusal] (ctor Refusal::Refusal.unsupportedProtocolVersion version versions)))
}

definition{
  name: versionGate,
  parameters: [version : String, versions : (List String)],
  result: (Option Refusal::Refusal),
  body: (gate (isSupported version versions) version versions)
}

definition{
  name: afterDecode,
  parameters: [
    request : JsonRpc::Request,
    decoded : (Result Methods::ClientRequest Refusal::ParamsProblem)
  ],
  result: (Result ValidRequest Refusal::Refusal),
  body: (match decoded
    | Result.error problem => (ctor Result.error [ValidRequest, Refusal::Refusal] (ctor Refusal::Refusal.invalidParams problem))
    | Result.ok typed => (ctor Result.ok [ValidRequest, Refusal::Refusal] (record ValidRequest {id := (project request id), request := typed})))
}

definition{
  name: afterVersion,
  parameters: [
    request : JsonRpc::Request,
    method : Methods::Method,
    meta : Meta::RequestMeta,
    gated : (Option Refusal::Refusal)
  ],
  result: (Result ValidRequest Refusal::Refusal),
  body: (match gated
    | Option.some refusal => (ctor Result.error [ValidRequest, Refusal::Refusal] refusal)
    | Option.none => (afterDecode request (Decode::decodeParams method meta (paramsObject (project request params)))))
}

definition{
  name: afterHeaders,
  parameters: [
    versions : (List String),
    request : JsonRpc::Request,
    method : Methods::Method,
    meta : Meta::RequestMeta,
    header : (Option Refusal::HeaderProblem)
  ],
  result: (Result ValidRequest Refusal::Refusal),
  body: (match header
    | Option.some problem => (ctor Result.error [ValidRequest, Refusal::Refusal] (ctor Refusal::Refusal.headerMismatch problem))
    | Option.none => (afterVersion request method meta (versionGate (project meta protocolVersion) versions)))
}

definition{
  name: afterMeta,
  parameters: [
    headers : (Option Headers),
    versions : (List String),
    request : JsonRpc::Request,
    method : Methods::Method,
    read : (Result Meta::RequestMeta Refusal::ParamsProblem)
  ],
  result: (Result ValidRequest Refusal::Refusal),
  body: (match read
    | Result.error problem => (ctor Result.error [ValidRequest, Refusal::Refusal] (ctor Refusal::Refusal.invalidParams problem))
    | Result.ok meta => (afterHeaders versions request method meta (headerCheck headers method (project request method) (project meta protocolVersion) (paramsObject (project request params)))))
}
