definition{
  name: byteOf,
  parameters: [value : Nat],
  result: Bytes,
  body: (match (prim slice (byteTable) value 1 : (Option Bytes))

definition{
  name: listToBytes,
  parameters: [built : Bytes, items : (List Nat)],
  result: Bytes,
  body: (match items
    | List.nil => built
    | List.cons head tail => (listToBytes (prim append built (byteOf head) : Bytes) tail)),
  executable: true,
  recursive_argument: items
}

definition{
  name: byteAt,
  parameters: [source : Bytes, position : Nat],
  result: Nat,
  body: (match (prim index source position : (Option UInt8))
    | Option.some octet => (prim to_nat octet : Nat)
    | Option.none => 0),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: bytesFrom,
  parameters: [source : Bytes, position : Nat, collected : (List Nat)],
  result: (List Nat),
  body: (match position
    | Nat.zero => collected
    | Nat.succ below => (bytesFrom source below (cons (byteAt source below) collected))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: position
}
