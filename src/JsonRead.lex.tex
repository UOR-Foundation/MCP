  name: settleStep,
  parameters: [machine : Machine, offset : Nat, byte : Nat],
  result: ReadStep,
  body: (match (project machine mode)
    | Mode.finished done => (if (isWhitespace byte) then (ctor ReadStep.moved machine) else (refuse (ctor JsonText::JsonRefusal.trailingContent offset)))
    | Mode.afterValue => (if (isWhitespace byte) then (ctor ReadStep.moved machine) else (if (byte =? 44) then (afterComma machine offset) else (if (byte =? 93) then (closeArray machine offset) else (if (byte =? 125) then (closeObject machine offset) else (refuse (ctor JsonText::JsonRefusal.unexpectedByte offset))))))
    | Mode.value => (refuse (ctor JsonText::JsonRefusal.unexpectedByte offset))
    | Mode.valueOrClose => (refuse (ctor JsonText::JsonRefusal.unexpectedByte offset))
    | Mode.keyOrClose => (refuse (ctor JsonText::JsonRefusal.unexpectedByte offset))
    | Mode.key => (refuse (ctor JsonText::JsonRefusal.unexpectedByte offset))
    | Mode.colon => (refuse (ctor JsonText::JsonRefusal.unexpectedByte offset))

    | Mode.literal which matched => (refuse (ctor JsonText::JsonRefusal.unexpectedByte offset))
    | Mode.number state => (refuse (ctor JsonText::JsonRefusal.unexpectedByte offset))),
  executable: true
}

  executable: true

definition{
  name: textStep,
  parameters: [
    machine : Machine,
    offset : Nat,
    byte : Nat,
    phase : StrPhase,
    collected : (List Nat),
    isKey : Bool
  ],
  result: ReadStep,
  body: (match phase
    | StrPhase.plain => (if (byte =? 34) then (finishText machine offset collected isKey) else (if (byte =? 92) then (moveTo machine (ctor Mode.text (ctor StrPhase.slash) collected isKey)) else (if (byte <? 32) then (refuse (ctor JsonText::JsonRefusal.controlInString offset)) else (moveTo machine (ctor Mode.text (ctor StrPhase.plain) (cons byte collected) isKey)))))
    | StrPhase.slash => (if (byte =? 117) then (moveTo machine (ctor Mode.text (ctor StrPhase.hex 0 0 0) collected isKey)) else (if ((escapeByte byte) =? 0) then (refuse (ctor JsonText::JsonRefusal.badEscape offset)) else (moveTo machine (ctor Mode.text (ctor StrPhase.plain) (cons (escapeByte byte) collected) isKey))))
    | StrPhase.hex count unit high => (if (15 <? (hexValue byte)) then (refuse (ctor JsonText::JsonRefusal.badEscape offset)) else (if ((count + 1) =? 4) then (unitDone machine offset ((prim multiply unit 16 : Nat) + (hexValue byte)) high collected isKey) else (moveTo machine (ctor Mode.text (ctor StrPhase.hex (count + 1) ((prim multiply unit 16 : Nat) + (hexValue byte)) high) collected isKey))))
    | StrPhase.wantSlash high => (if (byte =? 92) then (moveTo machine (ctor Mode.text (ctor StrPhase.wantU high) collected isKey)) else (refuse (ctor JsonText::JsonRefusal.loneSurrogate offset)))
    | StrPhase.wantU high => (if (byte =? 117) then (moveTo machine (ctor Mode.text (ctor StrPhase.hex 0 0 high) collected isKey)) else (refuse (ctor JsonText::JsonRefusal.loneSurrogate offset)))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: numberStep,
  parameters: [machine : Machine, offset : Nat, byte : Nat, state : NumState],
  result: ReadStep,
  body: (match (project state phase)
    | NumPhase.minus => (if (byte =? 48) then (withNumber machine state (ctor NumPhase.zero)) else (if (Utf8::inRange byte 49 57) then (addDigit machine offset byte state false) else (refuse (ctor JsonText::JsonRefusal.badNumber offset))))
    | NumPhase.zero => (if (byte =? 46) then (withNumber machine state (ctor NumPhase.dot)) else (if (isExponentMark byte) then (withNumber machine state (ctor NumPhase.expMark)) else (if (isDigit byte) then (refuse (ctor JsonText::JsonRefusal.badNumber offset)) else (finishNumber machine offset byte state))))
    | NumPhase.whole => (if (isDigit byte) then (addDigit machine offset byte state false) else (if (byte =? 46) then (withNumber machine state (ctor NumPhase.dot)) else (if (isExponentMark byte) then (withNumber machine state (ctor NumPhase.expMark)) else (finishNumber machine offset byte state))))
    | NumPhase.dot => (if (isDigit byte) then (addDigit machine offset byte state true) else (refuse (ctor JsonText::JsonRefusal.badNumber offset)))
    | NumPhase.frac => (if (isDigit byte) then (addDigit machine offset byte state true) else (if (isExponentMark byte) then (withNumber machine state (ctor NumPhase.expMark)) else (finishNumber machine offset byte state)))
    | NumPhase.expMark => (if (byte =? 43) then (moveTo machine (numberMode (project state negative) (ctor NumPhase.expSign) (project state mantissa) (project state significant) (project state fraction) false (0 : Int))) else (if (byte =? 45) then (moveTo machine (numberMode (project state negative) (ctor NumPhase.expSign) (project state mantissa) (project state significant) (project state fraction) true (0 : Int))) else (if (isDigit byte) then (addExponent machine offset byte (record NumState {negative := (project state negative), phase := (project state phase), mantissa := (project state mantissa), significant := (project state significant), fraction := (project state fraction), expNegative := false, exponent := (0 : Int)}) false) else (refuse (ctor JsonText::JsonRefusal.badNumber offset)))))
    | NumPhase.expSign => (if (isDigit byte) then (addExponent machine offset byte state (project state expNegative)) else (refuse (ctor JsonText::JsonRefusal.badNumber offset)))
    | NumPhase.expDigits => (if (isDigit byte) then (addExponent machine offset byte state (project state expNegative)) else (finishNumber machine offset byte state))),
  axioms: [propext],
  executable: true
}

definition{
  name: readStep,
  parameters: [machine : Machine, offset : Nat, byte : Nat],
  result: ReadStep,
  body: (match (project machine mode)
    | Mode.value => (if (isWhitespace byte) then (ctor ReadStep.moved machine) else (startValue machine offset byte))
    | Mode.valueOrClose => (if (isWhitespace byte) then (ctor ReadStep.moved machine) else (if (byte =? 93) then (closeArray machine offset) else (startValue machine offset byte)))
    | Mode.afterValue => (settleStep machine offset byte)
    | Mode.keyOrClose => (if (isWhitespace byte) then (ctor ReadStep.moved machine) else (if (byte =? 125) then (closeObject machine offset) else (if (byte =? 34) then (moveTo machine (ctor Mode.text (ctor StrPhase.plain) (nil Nat) true)) else (refuse (ctor JsonText::JsonRefusal.unexpectedByte offset)))))
    | Mode.key => (if (isWhitespace byte) then (ctor ReadStep.moved machine) else (if (byte =? 34) then (moveTo machine (ctor Mode.text (ctor StrPhase.plain) (nil Nat) true)) else (refuse (ctor JsonText::JsonRefusal.unexpectedByte offset))))
    | Mode.colon => (if (isWhitespace byte) then (ctor ReadStep.moved machine) else (if (byte =? 58) then (moveTo machine (ctor Mode.value)) else (refuse (ctor JsonText::JsonRefusal.unexpectedByte offset))))
    | Mode.text phase collected isKey => (textStep machine offset byte phase collected isKey)
    | Mode.literal which matched => (literalStep machine offset byte which matched)

    | Mode.value => (ctor Result.error [Json::Json, JsonText::JsonRefusal] (ctor JsonText::JsonRefusal.unexpectedEnd))

definition{
  name: readLoop,
  parameters: [machine : Machine, offset : Nat, bytes : (List Nat)],
  result: (Result Json::Json JsonText::JsonRefusal),
  body: (match bytes
    | List.nil => (match (readEnd machine offset)
      | Result.ok parsed => (ctor Result.ok [Json::Json, JsonText::JsonRefusal] parsed)
      | Result.error reason => (ctor Result.error [Json::Json, JsonText::JsonRefusal] reason))
    | List.cons byte rest => (match (readStep machine offset byte)
      | ReadStep.refused reason => (ctor Result.error [Json::Json, JsonText::JsonRefusal] reason)
      | ReadStep.moved next => (readLoop next (offset + 1) rest))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true,
  recursive_argument: bytes
}
