    | Option.none => (afterInput tool (walkLoop (admitBudget) (project tool name) (ctor Json::Json.object (project (project tool inputSchema) members)) (cons (ctor Visit.visit (ctor Json::Json.object (project (project tool inputSchema) members)) true (nil String)) (nil Visit)) (noAcc)))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: admitTool,
  parameters: [tool : Content::Tool],
  result: (Result (List Extension::ParamRule) AdmitFault),
  body: (match (nameFault (project tool name) (bytesOf (project tool name)))
    | Option.some fault => (ctor Result.error [(List Extension::ParamRule), AdmitFault] fault)
    | Option.none => (admitShape tool)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: duplicateOf,
  parameters: [tools : (List Content::Tool), seen : (List String)],
  result: (Option String),
  body: (match tools
    | List.nil => (ctor Option.none [String])
    | List.cons head rest => (if (seenBefore (project head name) seen) then (ctor Option.some [String] (project head name)) else (duplicateOf rest (cons (project head name) seen)))),
  executable: true,
  recursive_argument: tools
}

definition{
  name: catalogLoop,
  parameters: [tools : (List Content::Tool), rules : (List Extension::ParamRule)],
  result: (Result (List Extension::ParamRule) AdmitFault),
  body: (match tools
    | List.nil => (ctor Result.ok [(List Extension::ParamRule), AdmitFault] (JsonRead::reverseOnto [Extension::ParamRule] rules (nil Extension::ParamRule)))
    | List.cons head rest => (match (admitTool head)
      | Result.error fault => (ctor Result.error [(List Extension::ParamRule), AdmitFault] fault)
      | Result.ok found => (catalogLoop rest (JsonRead::reverseOnto [Extension::ParamRule] found rules)))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: tools
}

definition{
  name: duplicateFault,

  body: (match (duplicateOf tools (nil String))
    | Option.some name => (ctor Option.some [AdmitFault] (ctor AdmitFault.nameDuplicate name))

  executable: true
}

definition{

    | AdmitFault.dialectUnsupported tool => (prim join (cons "the schema dialect of tool" (cons tool (cons "is not JSON Schema 2020-12" (nil String)))) " " : String)

    | AdmitFault.headerNotText tool => (prim join (cons "tool" (cons tool (cons "has an x-mcp-header that is not text" (nil String)))) " " : String)

    | AdmitFault.headerUnreachable tool => (prim join (cons "tool" (cons tool (cons "has an x-mcp-header that is not reached through properties alone" (nil String)))) " " : String)
    | AdmitFault.headerType tool => (prim join (cons "tool" (cons tool (cons "has an x-mcp-header on a parameter that is not a string, integer or boolean" (nil String)))) " " : String)
    | AdmitFault.headerDuplicate tool header => (prim join (cons "tool" (cons tool (cons "repeats the x-mcp-header" (cons header (nil String))))) " " : String)),
  axioms: [propext],
  executable: true
}
