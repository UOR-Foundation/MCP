definition{
  name: loggingLevelOfWire,
  parameters: [wire : String],
  result: LoggingLevelRead,
  body: (if (prim equal wire "debug" : Bool) then (ctor LoggingLevelRead.recognized (ctor LoggingLevel.debug)) else (if (prim equal wire "info" : Bool) then (ctor LoggingLevelRead.recognized (ctor LoggingLevel.info)) else (if (prim equal wire "notice" : Bool) then (ctor LoggingLevelRead.recognized (ctor LoggingLevel.notice)) else (if (prim equal wire "warning" : Bool) then (ctor LoggingLevelRead.recognized (ctor LoggingLevel.warning)) else (if (prim equal wire "error" : Bool) then (ctor LoggingLevelRead.recognized (ctor LoggingLevel.error)) else (if (prim equal wire "critical" : Bool) then (ctor LoggingLevelRead.recognized (ctor LoggingLevel.critical)) else (if (prim equal wire "alert" : Bool) then (ctor LoggingLevelRead.recognized (ctor LoggingLevel.alert)) else (if (prim equal wire "emergency" : Bool) then (ctor LoggingLevelRead.recognized (ctor LoggingLevel.emergency)) else (ctor LoggingLevelRead.unrecognized))))))))),
  executable: true

  }
