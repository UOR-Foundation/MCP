definition{
  name: reqText,
  parameters: [key : String, object : Json::JsonObject],
  result: (Result String Refusal::ParamsProblem),
  body: (match (Json::lookup key object)
    | Json::Lookup.absent => (ctor Result.error [String, Refusal::ParamsProblem] (ctor Refusal::ParamsProblem.missingField key))
    | Json::Lookup.present value => (match (Json::asText value)
      | Option.none => (ctor Result.error [String, Refusal::ParamsProblem] (ctor Refusal::ParamsProblem.malformedField key))
      | Option.some text => (ctor Result.ok [String, Refusal::ParamsProblem] text))),
  executable: true
}

definition{
  name: textMembersOnto,
  parameters: [
    key : String,
    members : (List (String * Json::Json)),
    built : (List (String * String))
  ],
  result: (Result (List (String * String)) Refusal::ParamsProblem),
  body: (match members
    | List.nil => (ctor Result.ok [(List (String * String)), Refusal::ParamsProblem] built)
    | List.cons entry rest => (match (Json::asText (second entry))
      | Option.none => (ctor Result.error [(List (String * String)), Refusal::ParamsProblem] (ctor Refusal::ParamsProblem.malformedField key))
      | Option.some text => (textMembersOnto key rest (cons ((first entry), text) built)))),
  executable: true,
  recursive_argument: members
}

definition{
  name: textMembers,
  parameters: [key : String, members : (List (String * Json::Json))],
  result: (Result (List (String * String)) Refusal::ParamsProblem),
  body: (match (textMembersOnto key members (nil (String * String)))
    | Result.ok built => (ctor Result.ok [(List (String * String)), Refusal::ParamsProblem] (JsonRead::reverseOnto [(String * String)] built (nil (String * String))))
    | Result.error problem => (ctor Result.error [(List (String * String)), Refusal::ParamsProblem] problem)),
  executable: true
}

definition{
  name: optStringMap,
  parameters: [key : String, object : Json::JsonObject],
  result: (Result (Option (List (String * String))) Refusal::ParamsProblem),
  body: (match (Json::lookup key object)
    | Json::Lookup.absent => (ctor Result.ok [(Option (List (String * String))), Refusal::ParamsProblem] (ctor Option.none [(List (String * String))]))
    | Json::Lookup.present value => (match (Json::asObject value)
      | Option.none => (ctor Result.error [(Option (List (String * String))), Refusal::ParamsProblem] (ctor Refusal::ParamsProblem.malformedField key))
      | Option.some found => (match (textMembers key (project found members))
        | Result.ok pairs => (ctor Result.ok [(Option (List (String * String))), Refusal::ParamsProblem] (ctor Option.some [(List (String * String))] pairs))
        | Result.error problem => (ctor Result.error [(Option (List (String * String))), Refusal::ParamsProblem] problem)))),
  executable: true
}

definition{
  name: completionReference,
  parameters: [object : Json::JsonObject],
  result: (Result Methods::CompletionReference Refusal::ParamsProblem),
  body: (match (reqText "type" object)
    | Result.ok kind => (if (prim equal kind "ref/prompt" : Bool) then (match (reqText "name" object)
      | Result.ok name => (match (optText "title" object)
        | Result.ok title => (ctor Result.ok [Methods::CompletionReference, Refusal::ParamsProblem] (ctor Methods::CompletionReference.prompt (record Methods::PromptReference {name := name, title := title})))
        | Result.error problem => (ctor Result.error [Methods::CompletionReference, Refusal::ParamsProblem] problem))
      | Result.error problem => (ctor Result.error [Methods::CompletionReference, Refusal::ParamsProblem] problem)) else (if (prim equal kind "ref/resource" : Bool) then (match (reqText "uri" object)
      | Result.ok uri => (ctor Result.ok [Methods::CompletionReference, Refusal::ParamsProblem] (ctor Methods::CompletionReference.resourceTemplate uri))
      | Result.error problem => (ctor Result.error [Methods::CompletionReference, Refusal::ParamsProblem] problem)) else (ctor Result.error [Methods::CompletionReference, Refusal::ParamsProblem] (ctor Refusal::ParamsProblem.malformedField "type"))))
    | Result.error problem => (ctor Result.error [Methods::CompletionReference, Refusal::ParamsProblem] problem)),
  executable: true
}

definition{
  name: decodeComplete,
  parameters: [meta : Meta::RequestMeta, object : Json::JsonObject],
  result: (Result Methods::ClientRequest Refusal::ParamsProblem),
  body: (match (reqObject "ref" object)
    | Result.ok reference_object => (match (completionReference reference_object)
      | Result.ok reference => (match (reqObject "argument" object)
        | Result.ok argument_object => (match (completionArgument argument_object)
          | Result.ok argument => (match (completionContext object)
            | Result.ok context => (ctor Result.ok [Methods::ClientRequest, Refusal::ParamsProblem] (ctor Methods::ClientRequest.completionComplete (record Methods::CompleteParams {meta := meta, reference := reference, argument := argument, context := context})))
            | Result.error problem => (ctor Result.error [Methods::ClientRequest, Refusal::ParamsProblem] problem))
          | Result.error problem => (ctor Result.error [Methods::ClientRequest, Refusal::ParamsProblem] problem))
        | Result.error problem => (ctor Result.error [Methods::ClientRequest, Refusal::ParamsProblem] problem))
