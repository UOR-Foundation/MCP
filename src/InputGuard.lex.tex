definition{
  name: stateVerdict,
  parameters: [opened : (Option String)],
  result: (Result (Option String) Refusal::ProviderRefusal),
  body: (match opened
    | Option.none => (ctor Result.error [(Option String), Refusal::ProviderRefusal] (ctor Refusal::ProviderRefusal.invalidRequestState))
    | Option.some payload => (ctor Result.ok [(Option String), Refusal::ProviderRefusal] (ctor Option.some [String] payload))),
  executable: true
}

definition{
  name: openWith,
  parameters: [key : Bytes, method : String, name : String, state : (Option String)],
  result: (Result (Option String) Refusal::ProviderRefusal),
  body: (match state
    | Option.none => (ctor Result.ok [(Option String), Refusal::ProviderRefusal] (ctor Option.none [String]))
    | Option.some given => (stateVerdict (RequestState::openState key method name given))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: allObjects,
  parameters: [members : (List (String * Json::Json))],
  result: Bool,
  body: (match members
    | List.nil => true
    | List.cons entry rest => (match (Json::asObject (second entry))
      | Option.none => false
      | Option.some found => (allObjects rest))),
  executable: true,
  recursive_argument: members
}

definition{
  name: responsesOk,
  parameters: [responses : (Option Json::JsonObject)],
  result: Bool,
  body: (match responses
    | Option.none => true
    | Option.some given => (allObjects (project given members))),
  executable: true
}

definition{
  name: shapeVerdict,
  parameters: [opened : (Result (Option String) Refusal::ProviderRefusal), shaped : Bool],
  result: (Result (Option String) Refusal::ProviderRefusal),
  body: (match opened
    | Result.error refusal => (ctor Result.error [(Option String), Refusal::ProviderRefusal] refusal)
    | Result.ok payload => (if shaped then (ctor Result.ok [(Option String), Refusal::ProviderRefusal] payload) else (ctor Result.error [(Option String), Refusal::ProviderRefusal] (ctor Refusal::ProviderRefusal.invalidInputResponses)))),
  executable: true
}

definition{
  name: verified,
  parameters: [
    key : Bytes,
    method : String,
    name : String,
    state : (Option String),
    responses : (Option Json::JsonObject)
  ],
  result: (Result (Option String) Refusal::ProviderRefusal),
  body: (shapeVerdict (openWith key method name state) (responsesOk responses)),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: openTool,
  parameters: [key : Bytes, params : Methods::CallToolParams],
  result: (Result Methods::ClientRequest Refusal::ProviderRefusal),
  body: (match (verified key "tools/call" (project params name) (project params requestState) (project params inputResponses))
    | Result.error refusal => (ctor Result.error [Methods::ClientRequest, Refusal::ProviderRefusal] refusal)
    | Result.ok payload => (ctor Result.ok [Methods::ClientRequest, Refusal::ProviderRefusal] (ctor Methods::ClientRequest.toolsCall (record Methods::CallToolParams {meta := (project params meta), name := (project params name), arguments := (project params arguments), inputResponses := (project params inputResponses), requestState := payload})))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

    | Result.error refusal => (ctor Result.error [Methods::ClientRequest, Refusal::ProviderRefusal] refusal)

    | Result.ok payload => (ctor Result.ok [Methods::ClientRequest, Refusal::ProviderRefusal] (ctor Methods::ClientRequest.promptsGet (record Methods::GetPromptParams {meta := (project params meta), name := (project params name), arguments := (project params arguments), inputResponses := (project params inputResponses), requestState := payload})))),
  axioms: [Classical.choice, Quot.sound, propext],

definition{
  name: openRequest,
  parameters: [key : Bytes, request : Methods::ClientRequest],
  result: (Result Methods::ClientRequest Refusal::ProviderRefusal),
  body: (match request
    | Methods::ClientRequest.toolsCall params => (openTool key params)
    | Methods::ClientRequest.resourcesRead params => (openRead key params)
    | Methods::ClientRequest.promptsGet params => (openPrompt key params)
    | Methods::ClientRequest.serverDiscover params => (ctor Result.ok [Methods::ClientRequest, Refusal::ProviderRefusal] request)
    | Methods::ClientRequest.toolsList params => (ctor Result.ok [Methods::ClientRequest, Refusal::ProviderRefusal] request)
    | Methods::ClientRequest.resourcesList params => (ctor Result.ok [Methods::ClientRequest, Refusal::ProviderRefusal] request)
    | Methods::ClientRequest.resourcesTemplatesList params => (ctor Result.ok [Methods::ClientRequest, Refusal::ProviderRefusal] request)
    | Methods::ClientRequest.promptsList params => (ctor Result.ok [Methods::ClientRequest, Refusal::ProviderRefusal] request)
    | Methods::ClientRequest.completionComplete params => (ctor Result.ok [Methods::ClientRequest, Refusal::ProviderRefusal] request)
    | Methods::ClientRequest.tasksGet params => (ctor Result.ok [Methods::ClientRequest, Refusal::ProviderRefusal] request)
    | Methods::ClientRequest.tasksUpdate params => (ctor Result.ok [Methods::ClientRequest, Refusal::ProviderRefusal] request)
    | Methods::ClientRequest.tasksCancel params => (ctor Result.ok [Methods::ClientRequest, Refusal::ProviderRefusal] request)),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: nameOfRequest,
  parameters: [request : Methods::ClientRequest],
  result: String,
  body: (match request
    | Methods::ClientRequest.toolsCall params => (project params name)
    | Methods::ClientRequest.resourcesRead params => (project params uri)
    | Methods::ClientRequest.promptsGet params => (project params name)
    | Methods::ClientRequest.serverDiscover params => ""
    | Methods::ClientRequest.toolsList params => ""
    | Methods::ClientRequest.resourcesList params => ""
    | Methods::ClientRequest.resourcesTemplatesList params => ""
    | Methods::ClientRequest.promptsList params => ""
    | Methods::ClientRequest.completionComplete params => ""
    | Methods::ClientRequest.tasksGet params => ""
    | Methods::ClientRequest.tasksUpdate params => ""
    | Methods::ClientRequest.tasksCancel params => ""),
  executable: true
}

definition{
  name: hasContent,
  parameters: [needed : Input::InputRequiredResult],
  result: Bool,
  body: (match needed
    | Input::InputRequiredResult.requests pending => (match pending
      | List.nil => false
      | List.cons entry rest => true)
    | Input::InputRequiredResult.state token => true
    | Input::InputRequiredResult.requestsAndState pending token => true),
  executable: true
}

definition{
  name: sealed,
  parameters: [sealedState : (Option String), requests : (List (String * Input::InputRequest))],
  result: (Result Input::InputRequiredResult Refusal::ProviderRefusal),
  body: (match sealedState
    | Option.none => (ctor Result.error [Input::InputRequiredResult, Refusal::ProviderRefusal] (ctor Refusal::ProviderRefusal.requestStateKeyMissing))
    | Option.some token => (match requests
      | List.nil => (ctor Result.ok [Input::InputRequiredResult, Refusal::ProviderRefusal] (ctor Input::InputRequiredResult.state token))
      | List.cons entry rest => (ctor Result.ok [Input::InputRequiredResult, Refusal::ProviderRefusal] (ctor Input::InputRequiredResult.requestsAndState requests token)))),
  executable: true
}

definition{
  name: sealNeeded,
  parameters: [key : Bytes, method : String, name : String, needed : Input::InputRequiredResult],
  result: (Result Input::InputRequiredResult Refusal::ProviderRefusal),
  body: (match needed
    | Input::InputRequiredResult.requests pending => (ctor Result.ok [Input::InputRequiredResult, Refusal::ProviderRefusal] needed)
    | Input::InputRequiredResult.state token => (sealed (RequestState::sealState key method name token) (nil (String * Input::InputRequest)))
    | Input::InputRequiredResult.requestsAndState pending token => (sealed (RequestState::sealState key method name token) pending)),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: placed,
  parameters: [
    key : Bytes,
    method : Methods::Method,
    name : String,
    needed : Input::InputRequiredResult
  ],
  result: (Result Methods::ResultBody Refusal::ProviderRefusal),
  body: (if (Methods::supportsInputRequired method) then (if (hasContent needed) then (match (sealNeeded key (Methods::methodWire method) name needed)
    | Result.error refusal => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] refusal)
    | Result.ok protected => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Methods::ResultBody.inputRequired protected))) else (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Refusal::ProviderRefusal.emptyInputRequired))) else (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] (ctor Refusal::ProviderRefusal.misplacedInputRequired (Methods::methodWire method)))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: settleBody,
  parameters: [key : Bytes, method : Methods::Method, name : String, body : Methods::ResultBody],
  result: (Result Methods::ResultBody Refusal::ProviderRefusal),
  body: (match body
    | Methods::ResultBody.inputRequired needed => (placed key method name needed)
    | Methods::ResultBody.discover found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] body)
    | Methods::ResultBody.listTools found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] body)
    | Methods::ResultBody.callTool found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] body)
    | Methods::ResultBody.progressed found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] body)
    | Methods::ResultBody.listResources found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] body)
    | Methods::ResultBody.readResource found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] body)
    | Methods::ResultBody.listResourceTemplates found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] body)
    | Methods::ResultBody.listPrompts found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] body)
    | Methods::ResultBody.getPrompt found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] body)
    | Methods::ResultBody.complete found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] body)
    | Methods::ResultBody.getTask found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] body)
    | Methods::ResultBody.acknowledged => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] body)
    | Methods::ResultBody.taskCreated found => (ctor Result.ok [Methods::ResultBody, Refusal::ProviderRefusal] body)),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: settleAnswer,
  parameters: [
    key : Bytes,
    request : Methods::ClientRequest,
    answered : (Result Methods::ResultBody Refusal::ProviderRefusal)
  ],
  result: (Result Methods::ResultBody Refusal::ProviderRefusal),
  body: (match answered
    | Result.error refusal => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] refusal)
    | Result.ok body => (settleBody key (Methods::methodOfRequest request) (nameOfRequest request) body)),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: afterOpen,
  parameters: [
    key : Bytes,
    limit : Nat,
    used : Nat,
    answer : (Fn (Methods::ClientRequest) (Result Methods::ResultBody Refusal::ProviderRefusal)),
    request : Methods::ClientRequest,
    opened : (Result Methods::ClientRequest Refusal::ProviderRefusal)
  ],
  result: (Result Methods::ResultBody Refusal::ProviderRefusal),
  body: (match opened
    | Result.error refusal => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] refusal)
    | Result.ok admitted => (settleAnswer key request (Tools::gate limit used answer admitted))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{

    request : Methods::ClientRequest

definition{
  name: placedOk,
  parameters: [
    method : Methods::Method,
    outcome : (Result Methods::ResultBody Refusal::ProviderRefusal)
  ],
  result: Bool,
  body: (if (Methods::supportsInputRequired method) then true else (match outcome
    | Result.error refusal => true
    | Result.ok body => (not (Methods::isInputRequired body)))),
  executable: true
}
