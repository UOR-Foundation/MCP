definition{
  name: cleanLoop,
  parameters: [bytes : (List Nat), state : Nat, built : (List Nat)],
  result: (List Nat),
  body: (match bytes
    | List.nil => (flush state built)
    | List.cons byte rest => (let step : Out := (held state byte built); (cleanLoop rest (project step state) (project step built)))),
  executable: true,
  recursive_argument: bytes
}

definition{
  name: cleanText,
  parameters: [content : String],

    | Option.some cleaned => cleaned),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true

definition{
  name: rebuild,
  parameters: [fuel : Nat, phase : Phase, holes : (List Hole)],
  result: (Option Json::Json),
  body: (match fuel
    | Nat.zero => (ctor Option.none [Json::Json])
    | Nat.succ below => (match phase
      | Phase.down value => (match value
        | Json::Json.null => (rebuild below (ctor Phase.up value) holes)
        | Json::Json.bool flag => (rebuild below (ctor Phase.up value) holes)
        | Json::Json.number amount => (rebuild below (ctor Phase.up value) holes)
        | Json::Json.text content => (let cleaned : String := (cleanText content); (rebuild below (ctor Phase.up (ctor Json::Json.text cleaned)) holes))
        | Json::Json.array items => (match items
          | List.nil => (rebuild below (ctor Phase.up value) holes)
          | List.cons head tail => (rebuild below (ctor Phase.down head) (cons (ctor Hole.inArray (nil Json::Json) tail) holes)))
        | Json::Json.object entries => (match entries
          | List.nil => (rebuild below (ctor Phase.up value) holes)
          | List.cons entry tail => (let label : String := (cleanText (first entry)); (rebuild below (ctor Phase.down (second entry)) (cons (ctor Hole.inObject label (nil (String * Json::Json)) tail) holes)))))
      | Phase.up finished => (match holes
        | List.nil => (ctor Option.some [Json::Json] finished)
        | List.cons hole rest => (match hole
          | Hole.inArray done todo => (let grown : (List Json::Json) := (cons finished done); (match todo
            | List.nil => (rebuild below (ctor Phase.up (ctor Json::Json.array (JsonRead::reverseOnto [Json::Json] grown (nil Json::Json)))) rest)
            | List.cons head tail => (rebuild below (ctor Phase.down head) (cons (ctor Hole.inArray grown tail) rest))))
          | Hole.inObject label done todo => (let grown : (List (String * Json::Json)) := (cons (label, finished) done); (match todo
            | List.nil => (rebuild below (ctor Phase.up (ctor Json::Json.object (JsonRead::reverseOnto [(String * Json::Json)] grown (nil (String * Json::Json))))) rest)
            | List.cons entry tail => (let next : String := (cleanText (first entry)); (rebuild below (ctor Phase.down (second entry)) (cons (ctor Hole.inObject next grown tail) rest))))))))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true,
  recursive_argument: fuel
}
