inductive{
  name: IdRead,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: noId, fields: []},
    {name: gotId, fields: [JsonRpc::RequestId]},

definition{
  name: readId,
  parameters: [object : Json::JsonObject],
  result: IdRead,
  body: (match (Json::lookup "id" object)
    | Json::Lookup.absent => (ctor IdRead.noId)
    | Json::Lookup.present found => (match found
      | Json::Json.text content => (ctor IdRead.gotId (ctor JsonRpc::RequestId.text content))
      | Json::Json.number amount => (match (Param::wholeNumber amount)
        | Option.none => (ctor IdRead.badId)
        | Option.some whole => (ctor IdRead.gotId (ctor JsonRpc::RequestId.number whole)))
      | Json::Json.null => (ctor IdRead.badId)
      | Json::Json.bool flag => (ctor IdRead.badId)
      | Json::Json.array items => (ctor IdRead.badId)
      | Json::Json.object members => (ctor IdRead.badId))),
  axioms: [propext],
  executable: true
}

definition{
  name: isResponseShaped,
  parameters: [object : Json::JsonObject],
  result: Bool,
  body: ((Json::hasKey "result" object) || (Json::hasKey "error" object)),
  executable: true
}

definition{
  name: withoutMethod,
  parameters: [object : Json::JsonObject],
  result: InvalidReason,
  body: (if (isResponseShaped object) then (ctor InvalidReason.responseMessage) else (ctor InvalidReason.missingMethod)),
  executable: true
}

definition{
  name: invalid,
  parameters: [ident : (Option JsonRpc::RequestId), reason : InvalidReason],
  result: (Result JsonRpc::Request Fault),
  body: (ctor Result.error [JsonRpc::Request, Fault] (ctor Fault.invalidRequest ident reason)),
  executable: true
}

definition{
  name: withParams,
  parameters: [ident : JsonRpc::RequestId, method : String, object : Json::JsonObject],
  result: (Result JsonRpc::Request Fault),
  body: (match (Json::lookup "params" object)
    | Json::Lookup.absent => (ctor Result.ok [JsonRpc::Request, Fault] (record JsonRpc::Request {id := ident, method := method, params := (ctor Option.none [Json::JsonObject])}))
    | Json::Lookup.present found => (match (Json::asObject found)
      | Option.none => (invalid (ctor Option.some [JsonRpc::RequestId] ident) (ctor InvalidReason.paramsNotObject))
      | Option.some params => (ctor Result.ok [JsonRpc::Request, Fault] (record JsonRpc::Request {id := ident, method := method, params := (ctor Option.some [Json::JsonObject] params)})))),
  executable: true
}

definition{
  name: withMethod,
  parameters: [ident : JsonRpc::RequestId, object : Json::JsonObject],
  result: (Result JsonRpc::Request Fault),
  body: (match (Json::lookup "method" object)
    | Json::Lookup.absent => (invalid (ctor Option.some [JsonRpc::RequestId] ident) (withoutMethod object))
    | Json::Lookup.present found => (match (Json::asText found)
      | Option.none => (invalid (ctor Option.some [JsonRpc::RequestId] ident) (ctor InvalidReason.methodNotText))
      | Option.some method => (withParams ident method object))),
  executable: true
}

definition{
  name: withVersion,
  parameters: [ident : JsonRpc::RequestId, object : Json::JsonObject],
  result: (Result JsonRpc::Request Fault),
  body: (match (Json::lookup "jsonrpc" object)
    | Json::Lookup.absent => (invalid (ctor Option.some [JsonRpc::RequestId] ident) (ctor InvalidReason.wrongVersion))
    | Json::Lookup.present found => (match (Json::asText found)
      | Option.none => (invalid (ctor Option.some [JsonRpc::RequestId] ident) (ctor InvalidReason.wrongVersion))
      | Option.some version => (if (prim equal version (JsonRpc::jsonrpcVersion) : Bool) then (withMethod ident object) else (invalid (ctor Option.some [JsonRpc::RequestId] ident) (ctor InvalidReason.wrongVersion))))),
  executable: true
}

definition{
  name: withId,
  parameters: [object : Json::JsonObject],
  result: (Result JsonRpc::Request Fault),
  body: (match (readId object)
    | IdRead.badId => (invalid (ctor Option.none [JsonRpc::RequestId]) (ctor InvalidReason.badId))
    | IdRead.noId => (match (Json::lookup "method" object)
      | Json::Lookup.absent => (invalid (ctor Option.none [JsonRpc::RequestId]) (withoutMethod object))
      | Json::Lookup.present found => (invalid (ctor Option.none [JsonRpc::RequestId]) (if (isResponseShaped object) then (ctor InvalidReason.responseMessage) else (ctor InvalidReason.notification))))
    | IdRead.gotId identifier => (withVersion identifier object)),
  axioms: [propext],
  executable: true
}

definition{
  name: decodeRequest,
  parameters: [message : Json::Json],
  result: (Result JsonRpc::Request Fault),
  body: (match message
    | Json::Json.object members => (withId (record Json::JsonObject {members := members}))
    | Json::Json.array items => (invalid (ctor Option.none [JsonRpc::RequestId]) (ctor InvalidReason.batch))
    | Json::Json.null => (invalid (ctor Option.none [JsonRpc::RequestId]) (ctor InvalidReason.notAnObject))
    | Json::Json.bool flag => (invalid (ctor Option.none [JsonRpc::RequestId]) (ctor InvalidReason.notAnObject))
    | Json::Json.number amount => (invalid (ctor Option.none [JsonRpc::RequestId]) (ctor InvalidReason.notAnObject))
    | Json::Json.text content => (invalid (ctor Option.none [JsonRpc::RequestId]) (ctor InvalidReason.notAnObject))),
  axioms: [propext],
  executable: true
}

definition{
  name: reasonMessage,
  parameters: [reason : InvalidReason],
  result: String,
  body: (match reason
    | InvalidReason.notAnObject => "Invalid Request: the body is not a JSON object"
    | InvalidReason.batch => "Invalid Request: batches are not supported"
    | InvalidReason.wrongVersion => "Invalid Request: jsonrpc must be 2.0"
    | InvalidReason.missingMethod => "Invalid Request: method is required"
    | InvalidReason.methodNotText => "Invalid Request: method must be a string"
    | InvalidReason.badId => "Invalid Request: ident must be a string or an integer"
    | InvalidReason.paramsNotObject => "Invalid Request: params must be an object"
    | InvalidReason.notification => "Invalid Request: notifications are not accepted"
    | InvalidReason.responseMessage => "Invalid Request: the client must not send responses"),
  executable: true
}

theorem{
  name: an_error_without_an_id_has_no_id_member,
  parameters: [error : JsonRpc::ErrorObject],
  statement: ((Json::lookupIn "id" (errorMembers (ctor Option.none [JsonRpc::RequestId]) error)) = (ctor Json::Lookup.absent)),
  proof: reflexivity{}
}

theorem{
  name: an_error_with_an_id_carries_that_id,
  parameters: [ident : JsonRpc::RequestId, error : JsonRpc::ErrorObject],
  statement: ((Json::lookupIn "id" (errorMembers (ctor Option.some [JsonRpc::RequestId] ident) error)) = (ctor Json::Lookup.present (idJson ident))),
  proof: reflexivity{}
}

theorem{
  name: an_error_response_is_an_object_of_its_members,
  parameters: [ident : (Option JsonRpc::RequestId), error : JsonRpc::ErrorObject],
  statement: ((errorResponse ident error) = (ctor Json::Json.object (errorMembers ident error))),
  proof: reflexivity{}
}

theorem{
  name: a_progress_notification_has_no_id_member,
  parameters: [note : Methods::ProgressNote],
  statement: ((Json::lookupIn "id" (match (progressNotification note)
    | Json::Json.null => (nil (String * Json::Json))
    | Json::Json.bool flag => (nil (String * Json::Json))
    | Json::Json.number amount => (nil (String * Json::Json))
    | Json::Json.text content => (nil (String * Json::Json))
    | Json::Json.array items => (nil (String * Json::Json))
    | Json::Json.object members => members)) = (ctor Json::Lookup.absent)),
  proof: reflexivity{}
}
