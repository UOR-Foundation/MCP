definition{name: blockBytes, parameters: [], result: Nat, body: 64, executable: true}

definition{
  name: byteU,
  parameters: [source : Bytes, position : Nat],
  result: UInt8,
  body: (match (prim index source position : (Option UInt8))
    | Option.some octet => octet
    | Option.none => (0 : UInt8)),
  executable: true
}

definition{
  name: rawByte,
  parameters: [source : Bytes, position : Nat],
  result: UInt32,
  body: (match (prim index source position : (Option UInt8))
    | Option.some octet => (match (prim checked_convert octet : (Option UInt32))
      | Option.some wide => wide
      | Option.none => (0 : UInt32))
    | Option.none => (0 : UInt32)),
  axioms: [Quot.sound, propext],
  executable: true
}

  body: (let size : Nat := (prim length message : Nat); (let total : Nat := (prim multiply (prim quotient (size + 72) 64 0 : Nat) 64 : Nat); (second (prim iterate (fun [message, prefix, size, total] (state : (Nat * Sha256::Digest)) => (((first state) + 64), (Sha256::compress (second state) (blockFrom message size total (prim multiply (prefix + size) 8 : Nat) (first state))))) (prim quotient total 64 0 : Nat) (0, start) : (Nat * Sha256::Digest))))),
  axioms: [Quot.sound, propext],
