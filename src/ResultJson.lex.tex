  name: natInt,
  parameters: [count : Nat, total : Int],
  result: Int,
  body: (match count
    | Nat.zero => total
    | Nat.succ below => (natInt below (prim subtract total (-1 : Int) : Int))),
  executable: true,
  recursive_argument: count
}
