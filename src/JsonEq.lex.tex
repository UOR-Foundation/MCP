definition{
  name: pairStep,
  parameters: [left : Json::Json, right : Json::Json, rest : (List (Json::Json * Json::Json))],
  result: EqStep,
  body: (if ((kindOf left) =? (kindOf right)) then (match left
    | Json::Json.null => (ctor EqStep.advanced rest)
    | Json::Json.bool flag => (match (Json::asBool right)
      | Option.none => (ctor EqStep.differ)
      | Option.some other => (if (prim equal flag other : Bool) then (ctor EqStep.advanced rest) else (ctor EqStep.differ)))
    | Json::Json.number amount => (match (Json::asNumber right)
      | Option.none => (ctor EqStep.differ)
      | Option.some other => (if (Numeric::isSame (Numeric::compare amount other)) then (ctor EqStep.advanced rest) else (ctor EqStep.differ)))
    | Json::Json.text content => (match (Json::asText right)
      | Option.none => (ctor EqStep.differ)
      | Option.some other => (if (prim equal content other : Bool) then (ctor EqStep.advanced rest) else (ctor EqStep.differ)))
    | Json::Json.array items => (match (Json::asArray right)
      | Option.none => (ctor EqStep.differ)
      | Option.some others => (advance (pairItems items others (nil (Json::Json * Json::Json))) rest))
    | Json::Json.object members => (match (Json::asObject right)
      | Option.none => (ctor EqStep.differ)
      | Option.some others => (if ((memberCount members 0) =? (memberCount (project others members) 0)) then (advance (pairMembers members (project others members) (nil (Json::Json * Json::Json))) rest) else (ctor EqStep.differ)))) else (ctor EqStep.differ)),
  axioms: [propext],
  executable: true
}

definition{
  name: eqLoop,
  parameters: [fuel : Nat, pending : (List (Json::Json * Json::Json))],
  result: Same,
  body: (match fuel
    | Nat.zero => (ctor Same.exhausted)
    | Nat.succ below => (match pending
      | List.nil => (ctor Same.yes)
      | List.cons pair rest => (match (pairStep (first pair) (second pair) rest)
        | EqStep.differ => (ctor Same.no)
        | EqStep.advanced more => (eqLoop below more)))),
  axioms: [propext],
  executable: true,
  recursive_argument: fuel
}
