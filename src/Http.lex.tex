definition{
  name: sameName,
  parameters: [left : (List Nat), right : (List Nat)],
  result: Bool,
  body: (match left
    | List.nil => (match right | List.nil => true | List.cons y ys => false)
    | List.cons x xs => (match right
      | List.nil => false
      | List.cons y ys => (if ((lowerByte x) =? (lowerByte y)) then (sameName xs ys) else false))),
  executable: true,
  recursive_argument: left
}

definition{
  name: sameBytes,

    | List.cons x xs => (match right
      | List.nil => false
      | List.cons y ys => (if (x =? y) then (sameBytes xs ys) else false))),
  executable: true,
  recursive_argument: left

definition{
  name: scanFields,
  parameters: [fields : (List Field), wanted : (List Nat), seen : Seen],
  result: Seen,
  body: (match fields
    | List.nil => seen
    | List.cons head tail => (if (sameName (project head name) wanted) then (scanFields tail wanted (seenAgain seen (project head value))) else (scanFields tail wanted seen))),
  executable: true,
  recursive_argument: fields

definition{

  parameters: [bytes : (List Nat)],

  body: (match bytes

definition{
  name: readField,
  parameters: [fields : (List Field), wanted : (List Nat)],
  result: Reading,
  body: (match (scanFields fields wanted (ctor Seen.none))
    | Seen.none => (ctor Reading.absent)
    | Seen.many => (ctor Reading.repeated)
    | Seen.one only => (match (cleanValue only)
      | Option.none => (ctor Reading.malformed)
      | Option.some trimmed => (ctor Reading.value trimmed))),
  executable: true
}

definition{
  name: stripPrefix,
  parameters: [prefix : (List Nat), bytes : (List Nat)],
  result: (Option (List Nat)),
  body: (match prefix
    | List.nil => (ctor Option.some [(List Nat)] bytes)
    | List.cons wanted more => (match bytes
      | List.nil => (ctor Option.none [(List Nat)])
      | List.cons head tail => (if (head =? wanted) then (stripPrefix more tail) else (ctor Option.none [(List Nat)])))),
  executable: true,
  recursive_argument: prefix
}

definition{
  name: sentinelPayload,
  parameters: [bytes : (List Nat)],
  result: (Option (List Nat)),
  body: (match (stripPrefix (sentinelPrefix) bytes)
    | Option.none => (ctor Option.none [(List Nat)])
    | Option.some rest => (match (stripPrefix (sentinelSuffixReversed) (reverseOnto rest (nil Nat)))
      | Option.none => (ctor Option.none [(List Nat)])
      | Option.some reversed => (ctor Option.some [(List Nat)] (reverseOnto reversed (nil Nat))))),
  executable: true
}

definition{
  name: decodedReading,
  parameters: [reading : Reading],
  result: Reading,
  body: (match reading
    | Reading.value bytes => (match (sentinelPayload bytes)
      | Option.none => reading
      | Option.some payload => (match (Base64::decode payload)
        | Option.none => (ctor Reading.malformed)
        | Option.some decoded => (ctor Reading.value decoded)))
    | Reading.absent => reading
    | Reading.malformed => reading
    | Reading.repeated => reading),
  executable: true
}

definition{
  name: stdOf,
  parameters: [kind : Refusal::HeaderKind, reading : Reading],
  result: Std,
  body: (match reading
    | Reading.absent => (noStd)
    | Reading.malformed => (record Std {text := (ctor Option.none [String]), problem := (ctor Option.some [Refusal::HeaderProblem] (ctor Refusal::HeaderProblem.malformedValue kind))})
    | Reading.repeated => (record Std {text := (ctor Option.none [String]), problem := (ctor Option.some [Refusal::HeaderProblem] (ctor Refusal::HeaderProblem.repeatedField kind))})
    | Reading.value bytes => (match (textOf bytes)
      | Option.none => (record Std {text := (ctor Option.none [String]), problem := (ctor Option.some [Refusal::HeaderProblem] (ctor Refusal::HeaderProblem.malformedValue kind))})
      | Option.some content => (record Std {text := (ctor Option.some [String] content), problem := (ctor Option.none [Refusal::HeaderProblem])}))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

  name: paramSeen,

    | Reading.absent => (ctor Param::ParamSeen.absent)
    | Reading.malformed => (ctor Param::ParamSeen.malformed)
    | Reading.repeated => (ctor Param::ParamSeen.malformed)
    | Reading.value bytes => (match (textOf bytes)
      | Option.none => (ctor Param::ParamSeen.malformed)
      | Option.some content => (ctor Param::ParamSeen.value content))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}
