definition{
  name: readCompleted,
  parameters: [source : Bytes, position : Nat, count : Nat, built : (List HostEffect::Completed)],
  result: (Result ((List HostEffect::Completed) * Nat) Frame::FrameFault),
  body: (match count
    | Nat.zero => (ctor Result.ok [((List HostEffect::Completed) * Nat), Frame::FrameFault] ((JsonRead::reverseOnto [HostEffect::Completed] built (nil HostEffect::Completed)), position))
    | Nat.succ below => (match (Frame::u32At source position)
      | Option.none => (ctor Result.error [((List HostEffect::Completed) * Nat), Frame::FrameFault] (ctor Frame::FrameFault.truncated position))
      | Option.some tag => (match (Frame::chunkAt source (position + 4))
        | Option.none => (ctor Result.error [((List HostEffect::Completed) * Nat), Frame::FrameFault] (ctor Frame::FrameFault.truncated (position + 4)))
        | Option.some outcome => (readCompleted source (project outcome next) below (cons (record HostEffect::Completed {tag := tag, outcome := (project outcome bytes)}) built))))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: count
}

definition{
  name: closed,
  parameters: [
    ended : Bool,
    request : Bytes,
    completed : (List HostEffect::Completed),
    position : Nat
  ],
  result: (Result Turn Frame::FrameFault),
  body: (if ended then (ctor Result.ok [Turn, Frame::FrameFault] (record Turn {request := request, completed := completed})) else (ctor Result.error [Turn, Frame::FrameFault] (ctor Frame::FrameFault.trailing position))),
  executable: true
}

definition{
  name: finishTurn,
  parameters: [source : Bytes, request : Bytes, read : ((List HostEffect::Completed) * Nat)],
  result: (Result Turn Frame::FrameFault),
  body: (closed ((second read) =? (prim length source : Nat)) request (first read) (second read)),
  executable: true
}

definition{
  name: afterCount,
  parameters: [source : Bytes, request : Frame::Chunk, count : (Option Nat)],
  result: (Result Turn Frame::FrameFault),
  body: (match count
    | Option.none => (ctor Result.error [Turn, Frame::FrameFault] (ctor Frame::FrameFault.truncated (project request next)))
    | Option.some total => (match (readCompleted source ((project request next) + 4) total (nil HostEffect::Completed))
      | Result.error fault => (ctor Result.error [Turn, Frame::FrameFault] fault)
      | Result.ok read => (finishTurn source (project request bytes) read))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: turnAfter,
  parameters: [input : Bytes, found : (Option Frame::Chunk)],
  result: (Result Turn Frame::FrameFault),
  body: (match found
    | Option.none => (ctor Result.error [Turn, Frame::FrameFault] (ctor Frame::FrameFault.truncated 0))
    | Option.some request => (afterCount input request (Frame::u32At input (project request next)))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: readTurn,
  parameters: [input : Bytes],
  result: (Result Turn Frame::FrameFault),
  body: (turnAfter input (Frame::chunkAt input 0)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: stepList,
  parameters: [step : HostEffect::Step],
  result: (List Nat),
  body: (match step
    | HostEffect::Step.done response => (prim append (Frame::u32List 0) response : (List Nat))
    | HostEffect::Step.perform request => (prim append (Frame::u32List 1) (prim append (Frame::u32List (project request tag)) (JsonText::bytesToList (project request payload)) : (List Nat)) : (List Nat))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: effectOfProvider,
  parameters: [inner : Refusal::ProviderRefusal],
  result: (Option HostEffect::EffectRequest),
  body: (match inner
    | Refusal::ProviderRefusal.unknownTool name => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.unknownPrompt name => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.resourceNotFound uri => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.unknownTask taskId => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.invalidArguments reason => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.invalidCursor cursor => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.badToolOutput reason => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.rateLimited limit => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.missingCapability required => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.internal reason => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.invalidRequestState => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.invalidInputResponses => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.misplacedInputRequired method => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.emptyInputRequired => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.requestStateKeyMissing => (ctor Option.none [HostEffect::EffectRequest])
    | Refusal::ProviderRefusal.effectRequested request => (ctor Option.some [HostEffect::EffectRequest] request)),
  executable: true
}

definition{
  name: effectOf,
  parameters: [outcome : (Result Dispatch::Reply Refusal::Refusal)],
  result: (Option HostEffect::EffectRequest),
  body: (match outcome
    | Result.ok reply => (ctor Option.none [HostEffect::EffectRequest])
    | Result.error refusal => (match refusal
      | Refusal::Refusal.invalidParams problem => (ctor Option.none [HostEffect::EffectRequest])
      | Refusal::Refusal.unsupportedProtocolVersion requested supported => (ctor Option.none [HostEffect::EffectRequest])
      | Refusal::Refusal.headerMismatch problem => (ctor Option.none [HostEffect::EffectRequest])
      | Refusal::Refusal.methodNotFound method => (ctor Option.none [HostEffect::EffectRequest])
      | Refusal::Refusal.provider inner => (effectOfProvider inner))),
  executable: true
}

definition{
  name: performed,
  parameters: [
    host : Tasks::TaskHost,
    accepts : Bool,
    handled : Dispatch::Handled,
    request : HostEffect::EffectRequest
  ],
  result: HostEffect::Step,
  body: (if (HostEffect::tagFits (project request tag)) then (ctor HostEffect::Step.perform request) else (ctor HostEffect::Step.done (Frame::writeResponse (Serve::respond host accepts (record Dispatch::Handled {id := (project handled id), outcome := (ctor Result.error [Dispatch::Reply, Refusal::Refusal] (tagRefusal))}))))),

  executable: true
}

definition{
  name: storeRequest,

definition{
  name: storing,
  parameters: [
    info : Extension::ProviderInfo,
    stored : Bool,
    table : (List Nat),
    response : Http::HttpResponse

  result: HostEffect::Step,
  body: (if stored then (ctor HostEffect::Step.done (Frame::writeResponse (Serve::withoutTable response))) else (if (HostEffect::tagFits (project info taskStoreTag)) then (ctor HostEffect::Step.perform (storeRequest info table)) else (tagFault))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true

  name: finished,

    | Option.none => (ctor HostEffect::Step.done (Frame::writeResponse response))

  axioms: [Classical.choice, Quot.sound, propext],
  executable: true

definition{
  name: afterDecodeStep,
  parameters: [
    host : Tasks::TaskHost,
    info : Extension::ProviderInfo,
    answer : (Fn (Methods::ClientRequest) (Result Methods::ResultBody Refusal::ProviderRefusal)),
    stored : Bool,
    used : Nat,
    fields : (List Http::Field),
    decoded : (Result JsonRpc::Request RpcWire::Fault)
  ],
  result: HostEffect::Step,
  body: (match decoded
    | Result.error fault => (ctor HostEffect::Step.done (Frame::writeResponse (Serve::jsonResponse (Status::faultStatus fault) (RpcWire::errorResponse (RpcWire::faultId fault) (RpcWire::faultObject fault)))))
    | Result.ok rpc => (answerRequestStep host info answer stored used fields rpc)),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: afterParseStep,
  parameters: [
    host : Tasks::TaskHost,
    info : Extension::ProviderInfo,
    answer : (Fn (Methods::ClientRequest) (Result Methods::ResultBody Refusal::ProviderRefusal)),
    stored : Bool,
    used : Nat,
    fields : (List Http::Field),
    parsed : (Result Json::Json JsonText::JsonRefusal)
  ],
  result: HostEffect::Step,
  body: (match parsed
    | Result.error refusal => (ctor HostEffect::Step.done (Frame::writeResponse (Serve::jsonResponse (Status::badRequestStatus) (RpcWire::errorResponse (ctor Option.none [JsonRpc::RequestId]) (RpcWire::faultObject (ctor RpcWire::Fault.parseError))))))
    | Result.ok message => (afterDecodeStep host info answer stored used fields (RpcWire::decodeRequest message))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: serveOpenStep,
  parameters: [
    info : Extension::ProviderInfo,
    answerWith : (Fn (Tasks::TaskHost Methods::ClientRequest) (Result Methods::ResultBody Refusal::ProviderRefusal)),
    stored : Bool,
    request : Http::HttpRequest
  ],
  result: HostEffect::Step,
  body: (if (Http::isPost (project request method)) then (afterParseStep (project request host) (Serve::keyed info (project request stateKey)) (fun [answerWith, request] (typed : Methods::ClientRequest) => (apply answerWith (project request host) typed)) stored (project request toolCalls) (project request fields) (JsonRead::readJson (project request body))) else (ctor HostEffect::Step.done (Frame::writeResponse (Serve::notAllowedResponse)))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: serveGatedStep,
  parameters: [
    info : Extension::ProviderInfo,
    answerWith : (Fn (Tasks::TaskHost Methods::ClientRequest) (Result Methods::ResultBody Refusal::ProviderRefusal)),
    stored : Bool,
    request : Http::HttpRequest,
    gate : (Option Transport::TransportRefusal)
  ],
  result: HostEffect::Step,
  body: (match gate
    | Option.some refusal => (ctor HostEffect::Step.done (Frame::writeResponse (Serve::forbiddenResponse refusal)))
    | Option.none => (serveOpenStep info answerWith stored request)),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: stepOfTurn,
  parameters: [
    info : Extension::ProviderInfo,
    answerWith : (Fn (Asked) (Result Methods::ResultBody Refusal::ProviderRefusal)),
    turn : Turn,
    read : (Result Http::HttpRequest Frame::FrameFault)
  ],
  result: HostEffect::Step,
  body: (match read
    | Result.error fault => (unreadable)
    | Result.ok request => (serveRequestStep info (fun [answerWith, turn] (host : Tasks::TaskHost) (typed : Methods::ClientRequest) => (apply answerWith (record Asked {completed := (project turn completed), host := host, request := typed}))) (HostEffect::hasTag (project turn completed) (project info taskStoreTag)) request)),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: stepOf,
  parameters: [
    info : Extension::ProviderInfo,
    answerWith : (Fn (Asked) (Result Methods::ResultBody Refusal::ProviderRefusal)),
    input : Bytes
  ],
  result: HostEffect::Step,
  body: (match (readTurn input)
    | Result.error fault => (unreadable)
    | Result.ok turn => (stepOfTurn info answerWith turn (Frame::readRequest (project turn request)))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

  result: (List Nat),

  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}
