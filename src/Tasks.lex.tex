inductive{
  name: TaskStatus,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: working, fields: []},
    {name: input_required, fields: []},
    {name: completed, fields: []},
    {name: failed, fields: []},
    {name: cancelled, fields: []}
  ]
}

structure{
  name: TaskFields,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: taskId, type: String},
    {name: statusMessage, type: (Option String)},
    {name: createdAt, type: String},
    {name: lastUpdatedAt, type: String},
    {name: ttlMs, type: (Option Int)},
    {name: pollIntervalMs, type: (Option Int)}
  ]
}

inductive{
  name: DetailedTask,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: working, fields: [TaskFields]},
    {name: input_required, fields: [TaskFields, (List (String * Input::InputRequest))]},
    {name: completed, fields: [TaskFields, Json::JsonObject]},
    {name: failed, fields: [TaskFields, JsonRpc::ErrorObject]},
    {name: cancelled, fields: [TaskFields]}
  ]
}

definition{
  name: statusOf,
  parameters: [task : DetailedTask],
  result: TaskStatus,
  body: (match task
    | DetailedTask.working fields => (ctor TaskStatus.working)
    | DetailedTask.input_required fields requests => (ctor TaskStatus.input_required)
    | DetailedTask.completed fields result => (ctor TaskStatus.completed)
    | DetailedTask.failed fields error => (ctor TaskStatus.failed)
    | DetailedTask.cancelled fields => (ctor TaskStatus.cancelled)),
  executable: true
}

definition{
  name: fieldsOf,
  parameters: [task : DetailedTask],
  result: TaskFields,
  body: (match task
    | DetailedTask.working fields => fields
    | DetailedTask.input_required fields requests => fields
    | DetailedTask.completed fields result => fields
    | DetailedTask.failed fields error => fields
    | DetailedTask.cancelled fields => fields),
  executable: true
}

definition{
  name: isTerminal,
  parameters: [status : TaskStatus],
  result: Bool,
  body: (match status
    | TaskStatus.working => false
    | TaskStatus.input_required => false
    | TaskStatus.completed => true
    | TaskStatus.failed => true
    | TaskStatus.cancelled => true),
  executable: true
}

definition{
  name: taskStatusWire,
  parameters: [status : TaskStatus],
  result: String,
  body: (match status
    | TaskStatus.working => "working"
    | TaskStatus.input_required => "input_required"
    | TaskStatus.completed => "completed"
    | TaskStatus.failed => "failed"
    | TaskStatus.cancelled => "cancelled"),
  executable: true
}

definition{
  name: carriesResult,
  parameters: [task : DetailedTask],
  result: Bool,
  body: (match task
    | DetailedTask.working fields => false
    | DetailedTask.input_required fields requests => false
    | DetailedTask.completed fields result => true
    | DetailedTask.failed fields error => false
    | DetailedTask.cancelled fields => false),
  executable: true
}

theorem{
  name: only_a_completed_task_carries_a_result,
  parameters: [task : DetailedTask],
  statement: ((carriesResult task) = (match (statusOf task)
    | TaskStatus.completed => true
    | TaskStatus.working => false
    | TaskStatus.input_required => false
    | TaskStatus.failed => false
    | TaskStatus.cancelled => false)),
  proof: cases{
    branches: [
      {proof: reflexivity{}, binders: [fields], constructor: working},
      {proof: reflexivity{}, binders: [fields, requests], constructor: input_required},
      {proof: reflexivity{}, binders: [fields, result], constructor: completed},
      {proof: reflexivity{}, binders: [fields, error], constructor: failed},
      {proof: reflexivity{}, binders: [fields], constructor: cancelled}
    ],
    scrutinee: task
  }
}

theorem{
  name: terminal_statuses_are_completed_failed_cancelled,
  parameters: [],
  statement: ((((isTerminal (ctor TaskStatus.completed)) = true) /\ ((isTerminal (ctor TaskStatus.failed)) = true)) /\ ((((isTerminal (ctor TaskStatus.cancelled)) = true) /\ ((isTerminal (ctor TaskStatus.working)) = false)) /\ ((isTerminal (ctor TaskStatus.input_required)) = false))),
  proof: decide{}
}

definition{
  name: touch,
  parameters: [fields : TaskFields, now : String],
  result: TaskFields,
  body: (record TaskFields {taskId := (project fields taskId), statusMessage := (project fields statusMessage), createdAt := (project fields createdAt), lastUpdatedAt := now, ttlMs := (project fields ttlMs), pollIntervalMs := (project fields pollIntervalMs)}),
  executable: true
}

definition{
  name: due,
  parameters: [after : Int, createdMs : Int, nowMs : Int],
  result: Bool,
  body: (not (prim less_than (prim subtract nowMs createdMs : Int) after : Bool)),
  axioms: [propext],
  executable: true
}

definition{
  name: kept,
  parameters: [entry : TaskEntry],
  result: Stepped,
  body: (record Stepped {entry := entry, changed := false}),
  executable: true
}

definition{
  name: parkedPlan,
  parameters: [plan : TaskPlan],
  result: TaskPlan,
  body: (record TaskPlan {parkAfterMs := (project plan parkAfterMs), parked := (nil (String * Input::InputRequest)), finishAfterMs := (project plan finishAfterMs), outcome := (project plan outcome)}),
  executable: true
}

definition{
  name: concluded,
  parameters: [outcome : PlanOutcome, fields : TaskFields],
  result: DetailedTask,
  body: (match outcome
    | PlanOutcome.completes found => (ctor DetailedTask.completed fields found)
    | PlanOutcome.fails error => (ctor DetailedTask.failed fields error)),
  executable: true
}

definition{
  name: moved,
  parameters: [entry : TaskEntry, task : DetailedTask, plan : TaskPlan],
  result: Stepped,
  body: (record Stepped {entry := (record TaskEntry {task := task, createdMs := (project entry createdMs), plan := plan}), changed := true}),
  executable: true
}

definition{
  name: settleWorking,
  parameters: [entry : TaskEntry, fields : TaskFields, nowMs : Int, now : String],
  result: Stepped,
  body: (match (project (project entry plan) parked)
    | List.nil => (if (due (project (project entry plan) finishAfterMs) (project entry createdMs) nowMs) then (moved entry (concluded (project (project entry plan) outcome) (touch fields now)) (project entry plan)) else (kept entry))
    | List.cons head rest => (if (due (project (project entry plan) parkAfterMs) (project entry createdMs) nowMs) then (moved entry (ctor DetailedTask.input_required (touch fields now) (project (project entry plan) parked)) (parkedPlan (project entry plan))) else (kept entry))),
  axioms: [propext],
  executable: true
}

definition{
  name: settle,
  parameters: [entry : TaskEntry, nowMs : Int, now : String],
  result: Stepped,
  body: (match (project entry task)
    | DetailedTask.working fields => (settleWorking entry fields nowMs now)
    | DetailedTask.input_required fields requests => (kept entry)
    | DetailedTask.completed fields result => (kept entry)
    | DetailedTask.failed fields error => (kept entry)
    | DetailedTask.cancelled fields => (kept entry)),
  axioms: [propext],
  executable: true
}

definition{
  name: unansweredOnto,
  parameters: [
    requests : (List (String * Input::InputRequest)),
    answers : Json::JsonObject,
    built : (List (String * Input::InputRequest))
  ],
  result: (List (String * Input::InputRequest)),
  body: (match requests
    | List.nil => built
    | List.cons head rest => (if (Json::hasKey (first head) answers) then (unansweredOnto rest answers built) else (unansweredOnto rest answers (cons head built)))),
  executable: true,
  recursive_argument: requests
}

definition{
  name: unanswered,
  parameters: [requests : (List (String * Input::InputRequest)), answers : Json::JsonObject],
  result: (List (String * Input::InputRequest)),
  body: (JsonRead::reverseOnto [(String * Input::InputRequest)] (unansweredOnto requests answers (nil (String * Input::InputRequest))) (nil (String * Input::InputRequest))),
  executable: true
}

definition{
  name: resume,
  parameters: [entry : TaskEntry, fields : TaskFields, nowMs : Int, now : String],
  result: Stepped,
  body: (let resumed : TaskEntry := (record TaskEntry {task := (ctor DetailedTask.working (touch fields now)), createdMs := (project entry createdMs), plan := (project entry plan)}); (record Stepped {entry := (project (settle resumed nowMs now) entry), changed := true})),
  axioms: [propext],
  executable: true
}

definition{
  name: answerInput,
  parameters: [
    entry : TaskEntry,
    fields : TaskFields,
    requests : (List (String * Input::InputRequest)),
    answers : Json::JsonObject,
    nowMs : Int,
    now : String
  ],
  result: Stepped,
  body: (match (unanswered requests answers)
    | List.nil => (resume entry fields nowMs now)
    | List.cons head rest => (if ((prim length (cons head rest) : Nat) =? (prim length requests : Nat)) then (kept entry) else (moved entry (ctor DetailedTask.input_required (touch fields now) (cons head rest)) (project entry plan)))),
  axioms: [propext],
  executable: true
}

definition{
  name: answer,
  parameters: [entry : TaskEntry, answers : Json::JsonObject, nowMs : Int, now : String],
  result: Stepped,
  body: (match (project entry task)
    | DetailedTask.working fields => (kept entry)
    | DetailedTask.input_required fields requests => (answerInput entry fields requests answers nowMs now)
    | DetailedTask.completed fields result => (kept entry)
    | DetailedTask.failed fields error => (kept entry)
    | DetailedTask.cancelled fields => (kept entry)),
  axioms: [propext],
  executable: true
}

definition{
  name: cancelSettled,
  parameters: [settled : Stepped, now : String],
  result: Stepped,
  body: (match (project (project settled entry) task)
    | DetailedTask.working fields => (moved (project settled entry) (ctor DetailedTask.cancelled (touch fields now)) (project (project settled entry) plan))
    | DetailedTask.input_required fields requests => (moved (project settled entry) (ctor DetailedTask.cancelled (touch fields now)) (project (project settled entry) plan))
    | DetailedTask.completed fields result => settled
    | DetailedTask.failed fields error => settled
    | DetailedTask.cancelled fields => settled),
  executable: true
}

definition{
  name: cancel,
  parameters: [entry : TaskEntry, nowMs : Int, now : String],
  result: Stepped,
  body: (cancelSettled (settle entry nowMs now) now),
  axioms: [propext],
  executable: true
}

definition{
  name: idOf,
  parameters: [entry : TaskEntry],
  result: String,
  body: (project (fieldsOf (project entry task)) taskId),
  executable: true
}

definition{
  name: isExpired,
  parameters: [entry : TaskEntry, nowMs : Int],
  result: Bool,
  body: (match (project (fieldsOf (project entry task)) ttlMs)
    | Option.none => false
    | Option.some limit => (not (prim less_than (prim subtract nowMs (project entry createdMs) : Int) limit : Bool))),
  axioms: [propext],
  executable: true
}

definition{
  name: survivorsOnto,
  parameters: [wanted : String, entries : (List TaskEntry), nowMs : Int, built : (List TaskEntry)],
  result: (List TaskEntry),
  body: (match entries
    | List.nil => built
    | List.cons head rest => (if ((prim equal (idOf head) wanted : Bool) || (isExpired head nowMs)) then (survivorsOnto wanted rest nowMs built) else (survivorsOnto wanted rest nowMs (cons head built)))),
  axioms: [propext],
  executable: true,
  recursive_argument: entries
}

definition{
  name: withEntry,
  parameters: [entry : TaskEntry, table : (List TaskEntry), nowMs : Int],
  result: (List TaskEntry),
  body: (cons entry (JsonRead::reverseOnto [TaskEntry] (survivorsOnto (idOf entry) table nowMs (nil TaskEntry)) (nil TaskEntry))),
  axioms: [propext],
  executable: true
}

definition{
  name: findEntry,
  parameters: [wanted : String, table : (List TaskEntry), nowMs : Int],
  result: (Option TaskEntry),
  body: (match table
    | List.nil => (ctor Option.none [TaskEntry])
    | List.cons head rest => (if (prim equal (idOf head) wanted : Bool) then (if (isExpired head nowMs) then (ctor Option.none [TaskEntry]) else (ctor Option.some [TaskEntry] head)) else (findEntry wanted rest nowMs))),
  axioms: [propext],
  executable: true,
  recursive_argument: table
}

definition{
  name: notifies,
  parameters: [found : StoredTask],
  result: Bool,
  body: ((project found changed) && (match (project found shape)
    | StoredShape.read => true
    | StoredShape.acked => true)),
  executable: true
}
