  body: (match (Json::lookupIn "type" members)
    | Json::Lookup.absent => (ctor Kw.ok)
    | Json::Lookup.present declared => (match (Json::asText declared)
      | Option.some name => (if (validTypeName name) then (if (typeAllows name value) then (ctor Kw.ok) else (ctor Kw.fails)) else (ctor Kw.bad))
      | Option.none => (match (Json::asArray declared)
        | Option.some names => (typeList names value false)
        | Option.none => (ctor Kw.bad)))),
  axioms: [propext],
  executable: true

  result: Grown,
  body: (joinGrown (refFrames root members value path) (joinGrown (propertyFrames members value path) (joinGrown (extraFrames members value path) (joinGrown (itemFrames members value path) (joinGrown (allOfFrames members value path) (joinGrown (deferred (ctor Applicator.anyOfKind) (anyOfFrames members value path) members value path) (joinGrown (deferred (ctor Applicator.oneOfKind) (oneOfFrames members value path) members value path) (joinGrown (deferred (ctor Applicator.notKind) (notFrames members value path) members value path) (deferred (ctor Applicator.ifKind) (ifFrames members value path) members value path))))))))),
  axioms: [propext],
  executable: true
}

definition{
  name: grow,
  parameters: [
    root : Json::Json,
    members : (List (String * Json::Json)),
    value : Json::Json,
    path : (List Seg)
  ],
  result: Grown,
  body: (match (immediate members value)
    | Grown.failed keyword => (ctor Grown.failed keyword)
    | Grown.malformedKw keyword => (ctor Grown.malformedKw keyword)
    | Grown.unresolvedRef reference => (ctor Grown.unresolvedRef reference)
    | Grown.built clear => (structural root members value path)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: expand,
  parameters: [root : Json::Json, schema : Json::Json, value : Json::Json, path : (List Seg)],
  result: Grown,
  body: (match (Json::asBool schema)
    | Option.some flag => (if flag then (ctor Grown.built (nil Frame)) else (ctor Grown.failed "false"))
    | Option.none => (match (Json::asObject schema)
      | Option.some declared => (grow root (project declared members) value path)
      | Option.none => (ctor Grown.malformedKw "schema"))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: nextAny,
  parameters: [
    alternatives : (List Json::Json),
    value : Json::Json,
    path : (List Seg),
    rest : (List Frame)
  ],
  result: Next,
  body: (match alternatives
    | List.nil => (record Next {mode := (ctor Mode.failing path "anyOf"), stack := rest})
    | List.cons head tail => (record Next {mode := (ctor Mode.running), stack := (cons (ctor Frame.check head value path) (cons (ctor Frame.anyOf tail value path) rest))})),
  executable: true
}

definition{
  name: nextOne,
  parameters: [
    alternatives : (List Json::Json),
    value : Json::Json,
    path : (List Seg),
    found : Nat,
    rest : (List Frame)
  ],
  result: Next,
  body: (match alternatives
    | List.nil => (if (found =? 1) then (record Next {mode := (ctor Mode.running), stack := rest}) else (record Next {mode := (ctor Mode.failing path "oneOf"), stack := rest}))
    | List.cons head tail => (record Next {mode := (ctor Mode.running), stack := (cons (ctor Frame.check head value path) (cons (ctor Frame.oneOf tail value path found) rest))})),
  executable: true
}

definition{
  name: nextBranch,
  parameters: [
    branch : (Option Json::Json),
    value : Json::Json,
    path : (List Seg),
    rest : (List Frame)
  ],
  result: Next,
  body: (match branch
    | Option.none => (record Next {mode := (ctor Mode.running), stack := rest})
    | Option.some schema => (record Next {mode := (ctor Mode.running), stack := (cons (ctor Frame.check schema value path) rest)})),
  executable: true
}

definition{
  name: nextOnSuccess,
  parameters: [frame : Frame, rest : (List Frame)],
  result: Next,
  body: (match frame
    | Frame.check schema value path => (record Next {mode := (ctor Mode.running), stack := rest})
    | Frame.anyOf alternatives value path => (record Next {mode := (ctor Mode.running), stack := rest})
    | Frame.oneOf alternatives value path found => (nextOne alternatives value path (found + 1) rest)
    | Frame.notGuard path => (record Next {mode := (ctor Mode.failing path "not"), stack := rest})
    | Frame.ifGuard thenBranch elseBranch value path => (nextBranch thenBranch value path rest)
    | Frame.defer kind schema value path => (record Next {mode := (ctor Mode.running), stack := rest})),
  executable: true
}

definition{
  name: nextOnFailure,
  parameters: [frame : Frame, rest : (List Frame), at : (List Seg), keyword : String],
  result: Next,
  body: (match frame
    | Frame.check schema value path => (record Next {mode := (ctor Mode.failing at keyword), stack := rest})
    | Frame.anyOf alternatives value path => (nextAny alternatives value path rest)
    | Frame.oneOf alternatives value path found => (nextOne alternatives value path found rest)
    | Frame.notGuard path => (record Next {mode := (ctor Mode.running), stack := rest})
    | Frame.ifGuard thenBranch elseBranch value path => (nextBranch elseBranch value path rest)
    | Frame.defer kind schema value path => (record Next {mode := (ctor Mode.failing at keyword), stack := rest})),
  executable: true
}

definition{
  name: run,
  parameters: [root : Json::Json, fuel : Nat, mode : Mode, stack : (List Frame)],
  result: Verdict,
  body: (match fuel
    | Nat.zero => (ctor Verdict.exhausted)
    | Nat.succ below => (match stack
      | List.nil => (match mode
        | Mode.running => (ctor Verdict.valid)
        | Mode.failing path keyword => (ctor Verdict.invalid path keyword))
      | List.cons frame rest => (match mode
        | Mode.running => (match frame
          | Frame.check schema value path => (match (expand root schema value path)
            | Grown.failed keyword => (run root below (ctor Mode.failing path keyword) rest)
            | Grown.malformedKw keyword => (ctor Verdict.malformed keyword)
            | Grown.unresolvedRef reference => (ctor Verdict.unresolved reference)
            | Grown.built more => (let joined : (List Frame) := (prim append more rest : (List Frame)); (run root below mode joined)))
          | Frame.anyOf alternatives value path => (run root below mode rest)
          | Frame.oneOf alternatives value path found => (let next : Next := (nextOnSuccess frame rest); (run root below (project next mode) (project next stack)))
          | Frame.notGuard path => (run root below (ctor Mode.failing path "not") rest)
          | Frame.defer kind schema value path => (match (reopen kind schema value path)
            | Grown.failed keyword => (run root below (ctor Mode.failing path keyword) rest)
            | Grown.malformedKw keyword => (ctor Verdict.malformed keyword)
            | Grown.unresolvedRef reference => (ctor Verdict.unresolved reference)
            | Grown.built more => (let joined : (List Frame) := (prim append more rest : (List Frame)); (run root below mode joined)))
          | Frame.ifGuard thenBranch elseBranch value path => (let next : Next := (nextOnSuccess frame rest); (run root below (project next mode) (project next stack))))
        | Mode.failing at keyword => (let next : Next := (nextOnFailure frame rest at keyword); (run root below (project next mode) (project next stack)))))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: fuel
}

definition{
  name: validate,
  parameters: [root : Json::Json, schema : Json::Json, value : Json::Json],
  result: Verdict,
  body: (run root (stepBudget) (ctor Mode.running) (cons (ctor Frame.check schema value (nil Seg)) (nil Frame))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: escapeBytes,
  parameters: [bytes : (List Nat), built : (List Nat)],
  result: (List Nat),
  body: (match bytes
    | List.nil => built
    | List.cons byte rest => (if (byte =? 126) then (escapeBytes rest (cons 48 (cons 126 built))) else (if (byte =? 47) then (escapeBytes rest (cons 49 (cons 126 built))) else (escapeBytes rest (cons byte built))))),
  executable: true,
  recursive_argument: bytes
}

definition{
  name: escapeKey,
  parameters: [name : String],
  result: String,
  body: (match (prim utf8_decode (JsonText::listToBytes (JsonText::emptyBytes) (JsonRead::reverseOnto [Nat] (escapeBytes (JsonText::bytesToList (prim utf8_encode name : Bytes)) (nil Nat)) (nil Nat))) : (Option String))
    | Option.none => name
    | Option.some escaped => escaped),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: segText,
  parameters: [seg : Seg],
  result: String,

    | Seg.key name => (escapeKey name)
    | Seg.index position => (prim format_decimal (Numeric::natToInt position (0 : Int)) : String)),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{

  axioms: [Classical.choice, Quot.sound, propext],
  executable: true,
  recursive_argument: path
}
