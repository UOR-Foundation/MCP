\begin{semanticmodule}
definition{name: okStatus, parameters: [], result: Nat, body: 200, executable: true}

definition{name: badRequestStatus, parameters: [], result: Nat, body: 400, executable: true}

definition{name: notFoundStatus, parameters: [], result: Nat, body: 404, executable: true}

definition{name: tooManyRequestsStatus, parameters: [], result: Nat, body: 429, executable: true}

definition{name: methodNotAllowedStatus, parameters: [], result: Nat, body: 405, executable: true}

definition{name: serverErrorStatus, parameters: [], result: Nat, body: 500, executable: true}

definition{name: forbiddenStatus, parameters: [], result: Nat, body: 403, executable: true}

definition{
  name: transportStatus,
  parameters: [refusal : Transport::TransportRefusal],
  result: Nat,
  body: (match refusal
    | Transport::TransportRefusal.forbiddenOrigin => (forbiddenStatus)
    | Transport::TransportRefusal.forbiddenHost => (forbiddenStatus)),
  executable: true
}

definition{
  name: providerStatus,
  parameters: [inner : Refusal::ProviderRefusal],
  result: Nat,
  body: (match inner
    | Refusal::ProviderRefusal.unknownTool name => (badRequestStatus)
    | Refusal::ProviderRefusal.unknownPrompt name => (badRequestStatus)
    | Refusal::ProviderRefusal.resourceNotFound uri => (badRequestStatus)
    | Refusal::ProviderRefusal.unknownTask taskId => (badRequestStatus)
    | Refusal::ProviderRefusal.invalidArguments reason => (badRequestStatus)
    | Refusal::ProviderRefusal.invalidCursor cursor => (badRequestStatus)
    | Refusal::ProviderRefusal.badToolOutput reason => (serverErrorStatus)
    | Refusal::ProviderRefusal.rateLimited limit => (tooManyRequestsStatus)
    | Refusal::ProviderRefusal.missingCapability required => (badRequestStatus)
    | Refusal::ProviderRefusal.internal reason => (serverErrorStatus)
    | Refusal::ProviderRefusal.invalidRequestState => (badRequestStatus)
    | Refusal::ProviderRefusal.invalidInputResponses => (badRequestStatus)
    | Refusal::ProviderRefusal.misplacedInputRequired method => (serverErrorStatus)
    | Refusal::ProviderRefusal.emptyInputRequired => (serverErrorStatus)
    | Refusal::ProviderRefusal.requestStateKeyMissing => (serverErrorStatus)
    | Refusal::ProviderRefusal.effectRequested request => (serverErrorStatus)),
  executable: true
}

definition{
  name: statusOf,
  parameters: [refusal : Refusal::Refusal],
  result: Nat,
  body: (match refusal
    | Refusal::Refusal.invalidParams problem => (badRequestStatus)
    | Refusal::Refusal.unsupportedProtocolVersion requested supported => (badRequestStatus)
    | Refusal::Refusal.headerMismatch problem => (badRequestStatus)
    | Refusal::Refusal.methodNotFound method => (notFoundStatus)
    | Refusal::Refusal.provider inner => (providerStatus inner)),
  executable: true
}

definition{
  name: faultStatus,
  parameters: [fault : RpcWire::Fault],
  result: Nat,
  body: (match fault
    | RpcWire::Fault.parseError => (badRequestStatus)
    | RpcWire::Fault.invalidRequest ident reason => (badRequestStatus)),
  executable: true
}

definition{
  name: statusForCode,
  parameters: [code : Int],
  result: Nat,
  body: (if (Json::isZero (prim subtract code (JsonRpc::methodNotFoundCode) : Int)) then (notFoundStatus) else (if (Json::isZero (prim subtract code (JsonRpc::internalErrorCode) : Int)) then (serverErrorStatus) else (if (Json::isZero (prim subtract code (JsonRpc::rateLimitedCode) : Int)) then (tooManyRequestsStatus) else (badRequestStatus)))),
  axioms: [propext],
  executable: true
}

theorem{
  name: a_header_mismatch_is_a_bad_request,
  parameters: [problem : Refusal::HeaderProblem],
  statement: ((statusOf (ctor Refusal::Refusal.headerMismatch problem)) = 400),
  proof: reflexivity{}
}

theorem{
  name: a_fault_is_a_bad_request,
  parameters: [fault : RpcWire::Fault],
  statement: ((faultStatus fault) = 400),
  proof: cases{
    branches: [
      {proof: reflexivity{}, binders: [], constructor: parseError},
      {proof: reflexivity{}, binders: [ident, reason], constructor: invalidRequest}
    ],
    scrutinee: fault
  }
}

theorem{
  name: the_status_follows_the_code,
  parameters: [refusal : Refusal::Refusal],
  statement: ((statusOf refusal) = (statusForCode (Refusal::codeOf refusal))),
  proof: cases{
    branches: [
      {proof: reflexivity{}, binders: [problem], constructor: invalidParams},
      {
        proof: reflexivity{},
        binders: [requested, supported],
        constructor: unsupportedProtocolVersion
      },
      {proof: reflexivity{}, binders: [problem], constructor: headerMismatch},
      {proof: reflexivity{}, binders: [method], constructor: methodNotFound},
      {
        proof: cases{
          branches: [
            {proof: reflexivity{}, binders: [name], constructor: unknownTool},
            {proof: reflexivity{}, binders: [name], constructor: unknownPrompt},
            {proof: reflexivity{}, binders: [uri], constructor: resourceNotFound},
            {proof: reflexivity{}, binders: [taskId], constructor: unknownTask},
            {proof: reflexivity{}, binders: [reason], constructor: invalidArguments},
            {proof: reflexivity{}, binders: [cursor], constructor: invalidCursor},
            {proof: reflexivity{}, binders: [reason], constructor: badToolOutput},
            {proof: reflexivity{}, binders: [limit], constructor: rateLimited},
            {proof: reflexivity{}, binders: [required], constructor: missingCapability},
            {proof: reflexivity{}, binders: [reason], constructor: internal},
            {proof: reflexivity{}, binders: [], constructor: invalidRequestState},
            {proof: reflexivity{}, binders: [], constructor: invalidInputResponses},
            {proof: reflexivity{}, binders: [method], constructor: misplacedInputRequired},
            {proof: reflexivity{}, binders: [], constructor: emptyInputRequired},
            {proof: reflexivity{}, binders: [], constructor: requestStateKeyMissing},
            {proof: reflexivity{}, binders: [request], constructor: effectRequested}
          ],
          scrutinee: inner
        },
        binders: [inner],
        constructor: provider
      }
    ],
    scrutinee: refusal
  },
  axioms: [propext]
}
