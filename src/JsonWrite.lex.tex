inductive{
  name: WriteStep,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: failed, fields: [JsonText::JsonRefusal]},
    {name: advanced, fields: [(List Task), (List Nat)]}
  ]
}

definition{
  name: valueStep,
  parameters: [depth : Nat, item : Json::Json, rest : (List Task), collected : (List Nat)],
  result: WriteStep,
  body: (match item
    | Json::Json.null => (ctor WriteStep.advanced rest (JsonRead::reverseOnto [Nat] (literalText 2) collected))
    | Json::Json.bool flag => (ctor WriteStep.advanced rest (JsonRead::reverseOnto [Nat] (if flag then (literalText 0) else (literalText 1)) collected))
    | Json::Json.number amount => (if (inBounds amount) then (ctor WriteStep.advanced rest (JsonRead::reverseOnto [Nat] (numberBytes amount) collected)) else (ctor WriteStep.failed (ctor JsonText::JsonRefusal.encodeNumberOutOfRange)))
    | Json::Json.text content => (ctor WriteStep.advanced rest (JsonRead::reverseOnto [Nat] (quoted content) collected))
    | Json::Json.array elements => (if (depth <? (JsonRead::maxDepth)) then (ctor WriteStep.advanced (openTasks 91 (ctor Task.items (depth + 1) elements) rest) collected) else (ctor WriteStep.failed (ctor JsonText::JsonRefusal.encodeTooDeep)))
    | Json::Json.object entries => (if (depth <? (JsonRead::maxDepth)) then (ctor WriteStep.advanced (openTasks 123 (ctor Task.members (depth + 1) entries) rest) collected) else (ctor WriteStep.failed (ctor JsonText::JsonRefusal.encodeTooDeep)))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: taskStep,
  parameters: [task : Task, rest : (List Task), collected : (List Nat)],
  result: WriteStep,
  body: (match task
    | Task.emit pieces => (ctor WriteStep.advanced rest (JsonRead::reverseOnto [Nat] pieces collected))
    | Task.value depth item => (valueStep depth item rest collected)
    | Task.items depth elements => (match elements
      | List.nil => (ctor WriteStep.advanced (cons (ctor Task.emit (cons 93 (nil Nat))) rest) collected)
      | List.cons head others => (ctor WriteStep.advanced (cons (ctor Task.value depth head) (cons (ctor Task.afterItem depth others) rest)) collected))
    | Task.afterItem depth others => (match others
      | List.nil => (ctor WriteStep.advanced (cons (ctor Task.emit (cons 93 (nil Nat))) rest) collected)
      | List.cons next more => (ctor WriteStep.advanced (cons (ctor Task.emit (cons 44 (nil Nat))) (cons (ctor Task.items depth others) rest)) collected))
    | Task.members depth entries => (match entries
      | List.nil => (ctor WriteStep.advanced (cons (ctor Task.emit (cons 125 (nil Nat))) rest) collected)
      | List.cons entry others => (ctor WriteStep.advanced (cons (ctor Task.emit (prim append (quoted (first entry)) (cons 58 (nil Nat)) : (List Nat))) (cons (ctor Task.value depth (second entry)) (cons (ctor Task.afterMember depth others) rest))) collected))
    | Task.afterMember depth others => (match others
      | List.nil => (ctor WriteStep.advanced (cons (ctor Task.emit (cons 125 (nil Nat))) rest) collected)
      | List.cons next more => (ctor WriteStep.advanced (cons (ctor Task.emit (cons 44 (nil Nat))) (cons (ctor Task.members depth others) rest)) collected))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: writeLoop,
  parameters: [fuel : Nat, tasks : (List Task), collected : (List Nat)],
  result: (Result (List Nat) JsonText::JsonRefusal),
  body: (match fuel
    | Nat.zero => (ctor Result.error [(List Nat), JsonText::JsonRefusal] (ctor JsonText::JsonRefusal.encodeBudget))
    | Nat.succ below => (match tasks
      | List.nil => (ctor Result.ok [(List Nat), JsonText::JsonRefusal] collected)
      | List.cons task rest => (match (taskStep task rest collected)
        | WriteStep.failed reason => (ctor Result.error [(List Nat), JsonText::JsonRefusal] reason)
        | WriteStep.advanced pending produced => (writeLoop below pending produced)))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: fuel
}

definition{
  name: writeListWith,
  parameters: [budget : Nat, item : Json::Json],
  result: (Result (List Nat) JsonText::JsonRefusal),
  body: (match (writeLoop budget (cons (ctor Task.value 0 item) (nil Task)) (nil Nat))
    | Result.ok produced => (ctor Result.ok [(List Nat), JsonText::JsonRefusal] (JsonRead::reverseOnto [Nat] produced (nil Nat)))
    | Result.error reason => (ctor Result.error [(List Nat), JsonText::JsonRefusal] reason)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: writeList,
  parameters: [item : Json::Json],
  result: (Result (List Nat) JsonText::JsonRefusal),
  body: (writeListWith (writeBudget) item),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: writeJson,
  parameters: [item : Json::Json],
  result: (Result Bytes JsonText::JsonRefusal),
  body: (match (writeList item)
    | Result.ok produced => (ctor Result.ok [Bytes, JsonText::JsonRefusal] (JsonText::listToBytes (JsonText::emptyBytes) produced))
    | Result.error reason => (ctor Result.error [Bytes, JsonText::JsonRefusal] reason)),
  axioms: [Quot.sound, propext],
  executable: true
}
