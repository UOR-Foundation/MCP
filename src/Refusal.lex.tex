definition{
  name: providerCode,
  parameters: [refusal : ProviderRefusal],
  result: Int,
  body: (match refusal
    | ProviderRefusal.unknownTool name => (JsonRpc::invalidParamsCode)
    | ProviderRefusal.unknownPrompt name => (JsonRpc::invalidParamsCode)
    | ProviderRefusal.resourceNotFound uri => (JsonRpc::invalidParamsCode)
    | ProviderRefusal.unknownTask taskId => (JsonRpc::invalidParamsCode)
    | ProviderRefusal.invalidArguments reason => (JsonRpc::invalidParamsCode)
    | ProviderRefusal.missingCapability required => (JsonRpc::missingRequiredClientCapabilityCode)
    | ProviderRefusal.internal reason => (JsonRpc::internalErrorCode))
}

definition{
  name: codeOf,
  parameters: [refusal : Refusal],
  result: Int,
  body: (match refusal
    | Refusal.invalidParams problem => (JsonRpc::invalidParamsCode)
    | Refusal.unsupportedProtocolVersion requested supported => (JsonRpc::unsupportedProtocolVersionCode)
    | Refusal.headerMismatch problem => (JsonRpc::headerMismatchCode)
    | Refusal.methodNotFound method => (JsonRpc::methodNotFoundCode)
    | Refusal.provider inner => (providerCode inner))
}

definition{
  name: messageOfProblem,
  parameters: [problem : ParamsProblem],
  result: String,
  body: (match problem
    | ParamsProblem.missingParams => "Invalid params: params is required"
    | ParamsProblem.missingMeta => "Invalid params: _meta is required"
    | ParamsProblem.missingProtocolVersion => "Invalid params: _meta lacks io.modelcontextprotocol/protocolVersion"
    | ParamsProblem.missingClientCapabilities => "Invalid params: _meta lacks io.modelcontextprotocol/clientCapabilities"
    | ParamsProblem.malformedMeta => "Invalid params: _meta is not an object"
    | ParamsProblem.malformedProtocolVersion => "Invalid params: io.modelcontextprotocol/protocolVersion is not a string"
    | ParamsProblem.malformedClientCapabilities => "Invalid params: io.modelcontextprotocol/clientCapabilities is not a valid capabilities object"
    | ParamsProblem.missingField key => "Invalid params: a required field is missing"
    | ParamsProblem.malformedField key => "Invalid params: a field has the wrong type")
}

definition{
  name: messageOfHeader,

    | HeaderProblem.missingMethodHeader => "Header mismatch: Mcp-Method header is missing"
    | HeaderProblem.missingNameHeader => "Header mismatch: Mcp-Name header is missing"
    | HeaderProblem.protocolVersionDiffers => "Header mismatch: MCP-Protocol-Version differs from the request body"
    | HeaderProblem.methodDiffers => "Header mismatch: Mcp-Method differs from the request body"
    | HeaderProblem.nameDiffers => "Header mismatch: Mcp-Name differs from the request body")
}

definition{
  name: textList,
  parameters: [items : (List String)],
  result: (List Json::Json),
  body: (match items
    | List.nil => (nil Json::Json)
    | List.cons head tail => (cons (ctor Json::Json.text head) (textList tail))),
  recursive_argument: items
}

definition{
  name: optionalMember,
  parameters: [key : String, value : (Option Json::JsonObject)],
  result: (List (String * Json::Json)),
  body: (match value
    | Option.none => (nil (String * Json::Json))
    | Option.some declared => (cons (key, (ctor Json::Json.object (project declared members))) (nil (String * Json::Json))))
}

definition{
  name: capabilitiesJson,
  parameters: [capabilities : Capabilities::ClientCapabilities],
  result: Json::Json,
  body: (ctor Json::Json.object (prim append (optionalMember "experimental" (project capabilities experimental)) (prim append (optionalMember "roots" (project capabilities roots)) (prim append (optionalMember "sampling" (project capabilities sampling)) (prim append (optionalMember "elicitation" (project capabilities elicitation)) (optionalMember "extensions" (project capabilities extensions)) : (List (String * Json::Json))) : (List (String * Json::Json))) : (List (String * Json::Json))) : (List (String * Json::Json))))
}

definition{
  name: dataOf,
  parameters: [refusal : Refusal],
  result: (Option Json::Json),
  body: (match refusal
    | Refusal.invalidParams problem => (ctor Option.none [Json::Json])
    | Refusal.unsupportedProtocolVersion requested supported => (ctor Option.some [Json::Json] (ctor Json::Json.object (cons ("supported", (ctor Json::Json.array (textList supported))) (cons ("requested", (ctor Json::Json.text requested)) (nil (String * Json::Json))))))
    | Refusal.headerMismatch problem => (ctor Option.none [Json::Json])
    | Refusal.methodNotFound method => (ctor Option.none [Json::Json])
    | Refusal.provider inner => (match inner
      | ProviderRefusal.unknownTool name => (ctor Option.none [Json::Json])
      | ProviderRefusal.unknownPrompt name => (ctor Option.none [Json::Json])
      | ProviderRefusal.resourceNotFound uri => (ctor Option.none [Json::Json])
      | ProviderRefusal.unknownTask taskId => (ctor Option.none [Json::Json])
      | ProviderRefusal.invalidArguments reason => (ctor Option.none [Json::Json])
      | ProviderRefusal.missingCapability required => (ctor Option.some [Json::Json] (ctor Json::Json.object (cons ("requiredCapabilities", (capabilitiesJson required)) (nil (String * Json::Json)))))
      | ProviderRefusal.internal reason => (ctor Option.none [Json::Json])))
}

definition{
  name: messageOf,
  parameters: [refusal : Refusal],
  result: String,
  body: (match refusal
    | Refusal.invalidParams problem => (messageOfProblem problem)
    | Refusal.unsupportedProtocolVersion requested supported => "Unsupported protocol version"
    | Refusal.headerMismatch problem => (messageOfHeader problem)
    | Refusal.methodNotFound method => "Method not found"
    | Refusal.provider inner => (match inner
      | ProviderRefusal.unknownTool name => "Invalid params: unknown tool"
      | ProviderRefusal.unknownPrompt name => "Invalid params: unknown prompt"
      | ProviderRefusal.resourceNotFound uri => "Invalid params: resource not found"
      | ProviderRefusal.unknownTask taskId => "Invalid params: unknown task"
      | ProviderRefusal.invalidArguments reason => "Invalid params: invalid arguments"
      | ProviderRefusal.missingCapability required => "Missing required client capability"
      | ProviderRefusal.internal reason => "Internal error"))
}

definition{
  name: errorObjectOf,
  parameters: [refusal : Refusal],
  result: JsonRpc::ErrorObject,
  body: (record JsonRpc::ErrorObject {code := (codeOf refusal), message := (messageOf refusal), data := (dataOf refusal)})
}
