  name: a_done_step_is_the_word_zero_and_the_response,
  parameters: [response : (List Nat)],
  statement: ((StepServe::stepList (ctor HostEffect::Step.done response)) = (prim append (Frame::u32List 0) response : (List Nat))),
  proof: reflexivity{}
}

  statement: ((StepServe::readCompleted source position 0 built) = (ctor Result.ok [((List HostEffect::Completed) * Nat), Frame::FrameFault] ((JsonRead::reverseOnto [HostEffect::Completed] built (nil HostEffect::Completed)), position))),
  proof: reflexivity{}
}

theorem{
  name: an_effect_is_read_as_a_tag_and_a_length_prefixed_outcome,
  parameters: [source : Bytes, position : Nat, below : Nat, built : (List HostEffect::Completed)],
  statement: ((StepServe::readCompleted source position (below + 1) built) = (match (Frame::u32At source position)
    | Option.none => (ctor Result.error [((List HostEffect::Completed) * Nat), Frame::FrameFault] (ctor Frame::FrameFault.truncated position))
    | Option.some tag => (match (Frame::chunkAt source (position + 4))
      | Option.none => (ctor Result.error [((List HostEffect::Completed) * Nat), Frame::FrameFault] (ctor Frame::FrameFault.truncated (position + 4)))
      | Option.some outcome => (StepServe::readCompleted source (project outcome next) below (cons (record HostEffect::Completed {tag := tag, outcome := (project outcome bytes)}) built))))),
  proof: reflexivity{}
}
