\begin{lexlean}{Frame}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{mcp.core@1.0.0}
\importmodule{Http}
\importmodule{JsonRead}
\importmodule{JsonText}
\title{MCP exchange frames}
\begin{semanticmodule}
inductive{
  name: FrameFault,
  type_parameters: [],
  parameters: [],
  constructors: [{name: truncated, fields: [Nat]}, {name: trailing, fields: [Nat]}]
}

structure{
  name: Chunk,
  type_parameters: [],
  parameters: [],
  fields: [{name: bytes, type: Bytes}, {name: next, type: Nat}]
}

definition{
  name: u32At,
  parameters: [source : Bytes, position : Nat],
  result: (Option Nat),
  body: (if ((position + 4) <=? (prim length source : Nat)) then (ctor Option.some [Nat] ((prim multiply (JsonText::byteAt source position) 16777216 : Nat) + ((prim multiply (JsonText::byteAt source (position + 1)) 65536 : Nat) + ((prim multiply (JsonText::byteAt source (position + 2)) 256 : Nat) + (JsonText::byteAt source (position + 3)))))) else (ctor Option.none [Nat])),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: chunkAt,
  parameters: [source : Bytes, position : Nat],
  result: (Option Chunk),
  body: (match (u32At source position)
    | Option.none => (ctor Option.none [Chunk])
    | Option.some size => (match (prim slice source (position + 4) size : (Option Bytes))
      | Option.none => (ctor Option.none [Chunk])
      | Option.some taken => (ctor Option.some [Chunk] (record Chunk {bytes := taken, next := ((position + 4) + size)})))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: reverseFields,
  parameters: [fields : (List Http::Field), built : (List Http::Field)],
  result: (List Http::Field),
  body: (match fields
    | List.nil => built
    | List.cons head tail => (reverseFields tail (cons head built))),
  executable: true,
  recursive_argument: fields
}

definition{
  name: readFields,
  parameters: [source : Bytes, position : Nat, count : Nat, built : (List Http::Field)],
  result: (Result ((List Http::Field) * Nat) FrameFault),
  body: (match count
    | Nat.zero => (ctor Result.ok [((List Http::Field) * Nat), FrameFault] ((reverseFields built (nil Http::Field)), position))
    | Nat.succ below => (match (chunkAt source position)
      | Option.none => (ctor Result.error [((List Http::Field) * Nat), FrameFault] (ctor FrameFault.truncated position))
      | Option.some named => (match (chunkAt source (project named next))
        | Option.none => (ctor Result.error [((List Http::Field) * Nat), FrameFault] (ctor FrameFault.truncated (project named next)))
        | Option.some valued => (readFields source (project valued next) below (cons (record Http::Field {name := (JsonText::bytesToList (project named bytes)), value := (JsonText::bytesToList (project valued bytes))}) built))))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: count
}

definition{
  name: finishRequest,
  parameters: [source : Bytes, method : Chunk, target : Chunk, read : ((List Http::Field) * Nat)],
  result: (Result Http::HttpRequest FrameFault),
  body: (match (chunkAt source (second read))
    | Option.none => (ctor Result.error [Http::HttpRequest, FrameFault] (ctor FrameFault.truncated (second read)))
    | Option.some body => (if ((project body next) =? (prim length source : Nat)) then (ctor Result.ok [Http::HttpRequest, FrameFault] (record Http::HttpRequest {method := (JsonText::bytesToList (project method bytes)), target := (JsonText::bytesToList (project target bytes)), fields := (first read), body := (project body bytes)})) else (ctor Result.error [Http::HttpRequest, FrameFault] (ctor FrameFault.trailing (project body next))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: afterCount,
  parameters: [source : Bytes, method : Chunk, target : Chunk, count : (Option Nat)],
  result: (Result Http::HttpRequest FrameFault),
  body: (match count
    | Option.none => (ctor Result.error [Http::HttpRequest, FrameFault] (ctor FrameFault.truncated (project target next)))
    | Option.some total => (match (readFields source ((project target next) + 4) total (nil Http::Field))
      | Result.error fault => (ctor Result.error [Http::HttpRequest, FrameFault] fault)
      | Result.ok read => (finishRequest source method target read))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: afterTarget,
  parameters: [source : Bytes, method : Chunk, target : (Option Chunk)],
  result: (Result Http::HttpRequest FrameFault),
  body: (match target
    | Option.none => (ctor Result.error [Http::HttpRequest, FrameFault] (ctor FrameFault.truncated (project method next)))
    | Option.some found => (afterCount source method found (u32At source (project found next)))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: readRequest,
  parameters: [source : Bytes],
  result: (Result Http::HttpRequest FrameFault),
  body: (match (chunkAt source 0)
    | Option.none => (ctor Result.error [Http::HttpRequest, FrameFault] (ctor FrameFault.truncated 0))
    | Option.some method => (afterTarget source method (chunkAt source (project method next)))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: u32List,
  parameters: [value : Nat],
  result: (List Nat),
  body: (cons (prim remainder (prim quotient value 16777216 0 : Nat) 256 0 : Nat) (cons (prim remainder (prim quotient value 65536 0 : Nat) 256 0 : Nat) (cons (prim remainder (prim quotient value 256 0 : Nat) 256 0 : Nat) (cons (prim remainder value 256 0 : Nat) (nil Nat))))),
  executable: true
}

definition{
  name: fieldList,
  parameters: [field : Http::Field],
  result: (List Nat),
  body: (prim append (u32List (prim length (project field name) : Nat)) (prim append (project field name) (prim append (u32List (prim length (project field value) : Nat)) (project field value) : (List Nat)) : (List Nat)) : (List Nat)),
  executable: true
}

definition{
  name: fieldsListOnto,
  parameters: [reversed : (List Http::Field), built : (List Nat)],
  result: (List Nat),
  body: (match reversed
    | List.nil => built
    | List.cons head tail => (fieldsListOnto tail (prim append (fieldList head) built : (List Nat)))),
  executable: true,
  recursive_argument: reversed
}

definition{
  name: fieldsList,
  parameters: [fields : (List Http::Field)],
  result: (List Nat),
  body: (fieldsListOnto (JsonRead::reverseOnto [Http::Field] fields (nil Http::Field)) (nil Nat)),
  executable: true
}

definition{
  name: writeResponse,
  parameters: [response : Http::HttpResponse],
  result: (List Nat),
  body: (prim append (u32List (project response status)) (prim append (u32List (prim length (project response fields) : Nat)) (prim append (fieldsList (project response fields)) (prim append (u32List (prim length (project response body) : Nat)) (project response body) : (List Nat)) : (List Nat)) : (List Nat)) : (List Nat)),
  executable: true
}
\end{semanticmodule}
\end{lexlean}
